25 /// <summary>Enters the cost estimate: repair, material and equipment; total calculated (BUD-01).</summary>

33 /// <summary>Submits the estimate to Budget; status becomes Pending (BUD-02). 502 when Budget is unavailable (retry later).</summary>
34 /// <param name="id">Intervention ID.</param>
35 /// <returns>The HTTP result (status codes described above).</returns>
36 [HttpPost("api/interventions/{id:int}/budget/submit")]
37 public IActionResult Submit(int id)
38 {
39 var outcome = _service.Submit(id, RequestContext.Actor(Request));
40 return this.ToResult(outcome, i => Ok(new { alreadySubmitted = outcome.Duplicate, budget = SummaryOf(i) }));
41 }
42
43 /// <summary>Budget summary for an intervention: estimate, status, reference, decision, comparison.</summary>
44 /// <param name="id">Intervention ID.</param>
45 /// <returns>The HTTP result (status codes described above).</returns>
46 [HttpGet("api/interventions/{id:int}/budget")]
47 public IActionResult Get(int id)
48 {
49 var comparison = _service.Compare(id);
50 return this.ToResult(comparison, c => Ok(new

52 comparison = c,
53 requireApprovalToStart = _policy.RequireApprovalToStart
54 }));
55 }

57 /// <summary>Estimated vs actual cost and the difference (BUD-05).</summary>

59 /// <returns>The HTTP result (status codes described above).</returns>
60 [HttpGet("api/interventions/{id:int}/budget/comparison")]
61 public IActionResult Compare(int id) => this.ToResult(_service.Compare(id));

63 /// <summary>
64 /// Receives a decision from the Budget service (BUD-03). Requires header X-Source-Service: Budget.

72 var outcome = _service.ReceiveDecision(request, RequestContext.SourceService(Request));

76 /// <summary>Budget policy (BUD-04 setting).</summary>
77 /// <returns>The HTTP result (status codes described above).</returns>
78 [HttpGet("api/budget/policy")]

80
81 private IActionResult Summary(Intervention i) => Ok(SummaryOf(i));

83 private static object SummaryOf(Intervention i) => new

88 i.CostEstimate.RepairCost,

90 i.CostEstimate.EquipmentCost,
91 i.CostEstimate.Total,
92 i.CostEstimate.EstimatedBy,
93 i.CostEstimate.EstimatedAt
94 },
95 budgetStatus = i.BudgetStatus,
96 budgetRequestReference = i.BudgetRequestReference,
97 budgetDecision = i.BudgetDecision
98 };

16/// <summary>
17/// Tracks every exchange with external services and supports retries (TEC-04).
18/// - Outbound: each message has an idempotency key; a message that already succeeded is
19/// never sent again, so retries and double-clicks do not create duplicate requests or updates.
20/// - Failed outbound messages stay visible with their error and can be retried.
21/// - Inbound: each message is recorded under the sender's external ID so a repeated
22/// delivery is detected and not processed twice.
23/// </summary>
24public class ExchangeService
25{
26 private readonly JsonListStore<ExternalExchange> _store;
27 private readonly ExternalServiceHub _hub;
28
29 /// <summary>
30 /// Raised when an outbound exchange succeeds (first send or retry), so the service that
31 /// owns the business record (e.g. Budget, Security) can store the external reference.
32 /// </summary>
33 public event Action<ExternalExchange>? Succeeded;
34
35 /// <summary>Creates the service.</summary>
36 /// <param name="hub">Clients for the external services.</param>
37 /// <param name="filePath">JSON file, or null for memory only.</param>
38 public ExchangeService(ExternalServiceHub hub, string? filePath = null)
39 {
40 _hub = hub;
41 _store = new JsonListStore<ExternalExchange>(filePath);
42 }
43
44 /// <summary>
45 /// Sends a message unless the same message (same service, operation and key) already
46 /// succeeded. A previously failed exchange with the same key is re-attempted in place.
47 /// </summary>
48 /// <param name="service">External service name.</param>
49 /// <param name="operation">Contract operation.</param>
50 /// <param name="key">Idempotency key.</param>
51 /// <param name="payloadJson">Message body.</param>
52 /// <param name="interventionId">Related intervention, if any.</param>
53 /// <returns>The exchange and whether it had already been sent.</returns>
54 public SendOutcome Send(string service, string operation, string key, string payloadJson, int? interventionId)
55 {
56 lock (_store.Sync)
57 {
58 var existing = Find(service, ExchangeRules.Outbound, operation, key);
59 if (existing is { Status: ExchangeRules.Succeeded })
60 {
61 return new SendOutcome(existing, true);
62 }
63
64 var exchange = existing ?? new ExternalExchange
65 {
66 Id = _store.NextId(x => x.Id),
67 Service = service,
68 Direction = ExchangeRules.Outbound,
69 Operation = operation,
70 Key = key,
71 InterventionId = interventionId,
72 PayloadJson = payloadJson,
73 CreatedAt = DateTime.UtcNow
74 };
75 if (existing is null) _store.Items.Add(exchange);
76
77 Attempt(exchange);
78 _store.Save();
79 return new SendOutcome(exchange, false);
80 }
81 }
82
83 /// <summary>Retries a failed outbound exchange with its stored message.</summary>
84 /// <param name="id">Exchange ID.</param>
85 /// <returns>The outcome, or null when the exchange does not exist or is inbound.</returns>
86 public SendOutcome? Retry(int id)
87 {
88 lock (_store.Sync)
89 {
90 var exchange = _store.Items.FirstOrDefault(x => x.Id == id);
91 if (exchange is null || exchange.Direction != ExchangeRules.Outbound) return null;
92 if (exchange.Status == ExchangeRules.Succeeded) return new SendOutcome(exchange, true);
93
94 Attempt(exchange);
95 _store.Save();
96 return new SendOutcome(exchange, false);
97 }
98 }
99
100 /// <summary>
101 /// Records an inbound message. Returns the existing record and duplicate = true when
102 /// the same service already delivered a message with this external ID.
103 /// </summary>
104 /// <param name="service">Sending service.</param>
105 /// <param name="operation">Contract operation.</param>
106 /// <param name="externalId">Sender's ID for the message.</param>
107 /// <param name="payloadJson">Message body.</param>
108 /// <param name="interventionId">Related intervention, if any.</param>
109 /// <returns>The exchange and whether it was a duplicate delivery.</returns>
110 public (ExternalExchange Exchange, bool Duplicate) RecordInbound(
111 string service, string operation, string externalId, string payloadJson, int? interventionId = null)
112 {
113 lock (_store.Sync)
114 {
115 var existing = Find(service, ExchangeRules.Inbound, operation, externalId);
116 if (existing is not null) return (existing, true);
117
118 var exchange = new ExternalExchange
119 {
120 Id = _store.NextId(x => x.Id),
121 Service = service,
122 Direction = ExchangeRules.Inbound,
123 Operation = operation,
124 Key = externalId,
125 InterventionId = interventionId,
126 PayloadJson = payloadJson,
127 Status = ExchangeRules.Received,
128 Attempts = 1,
129 CreatedAt = DateTime.UtcNow,
130 LastAttemptAt = DateTime.UtcNow
131 };
132 _store.Items.Add(exchange);
133 _store.Save();
134 return (exchange, false);
135 }
136 }

138 /// <summary>Lists exchanges, newest first, with optional filters.</summary>
139 /// <param name="service">External service name, see ExternalServiceNames.</param>
140 /// <param name="status">Optional status filter.</param>
141 /// <param name="direction">Optional direction filter: Inbound or Outbound.</param>
142 /// <param name="interventionId">Intervention ID.</param>
143 /// <returns>The matching records.</returns>
144 public List<ExternalExchange> Query(string? service = null, string? status = null, string? direction = null, int? interventionId = null)
145 {
146 lock (_store.Sync)
147 {
148 return _store.Items
149 .Where(x => string.IsNullOrWhiteSpace(service) || string.Equals(x.Service, service, StringComparison.OrdinalIgnoreCase))
150 .Where(x => string.IsNullOrWhiteSpace(status) || string.Equals(x.Status, status, StringComparison.OrdinalIgnoreCase))
151 .Where(x => string.IsNullOrWhiteSpace(direction) || string.Equals(x.Direction, direction, StringComparison.OrdinalIgnoreCase))
152 .Where(x => interventionId is null || x.InterventionId == interventionId)
153 .OrderByDescending(x => x.CreatedAt).ThenByDescending(x => x.Id)
154 .ToList();
155 }
156 }

158 /// <summary>Gets one exchange.</summary>
159 /// <param name="id">Exchange ID.</param>

161 public ExternalExchange? GetById(int id)

163 lock (_store.Sync) { return _store.Items.FirstOrDefault(x => x.Id == id); }

166 private ExternalExchange? Find(string service, string direction, string operation, string key) =>
167 _store.Items.FirstOrDefault(x =>
168 string.Equals(x.Service, service, StringComparison.OrdinalIgnoreCase)
169 && x.Direction == direction
170 && x.Operation == operation
171 && x.Key == key);

173 private void Attempt(ExternalExchange exchange)
174 {
175 exchange.Attempts++;
176 exchange.LastAttemptAt = DateTime.UtcNow;
177
178 var client = _hub.Get(exchange.Service);
179 var result = client is null
180 ? new ExternalCallResult(false, null, $"No client configured for {exchange.Service}.")
181 : SafeSend(client, exchange);
182
183 exchange.Status = result.Success ? ExchangeRules.Succeeded : ExchangeRules.Failed;
184 exchange.ExternalReference = result.Success ? result.Reference : exchange.ExternalReference;
185 exchange.LastError = result.Success ? null : result.Error;
186
187 if (result.Success) Succeeded?.Invoke(exchange);
188 }
189
190 private static ExternalCallResult SafeSend(IExternalServiceClient client, ExternalExchange exchange)
191 {
192 try
193 {
194 return client.Send(exchange.Operation, exchange.PayloadJson);
195 }
196 catch (Exception ex)
197 {
198 return new ExternalCallResult(false, null, ex.Message);
199 }
200 }

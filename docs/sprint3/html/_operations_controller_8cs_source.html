50 /// <summary>Retries a failed outbound exchange (TEC-04). A succeeded exchange is never sent twice.</summary>
51 /// <param name="id">Exchange ID.</param>
52 /// <returns>The HTTP result (status codes described above).</returns>
53 [HttpPost("api/exchanges/{id:int}/retry")]
54 public IActionResult Retry(int id)
55 {
56 var outcome = _exchanges.Retry(id);
57 if (outcome is null) return NotFound(new { errors = new[] { "Outbound exchange not found." } });
58 return Ok(new { alreadySent = outcome.AlreadySent, succeeded = outcome.Succeeded, exchange = outcome.Exchange });
59 }

74 /// <summary>Turns a mock service on or off to test failures and retries (TEC-03, development only).</summary>
75 /// <param name="name">Mock service name, e.g. Budget.</param>
76 /// <param name="request">True to bring the mock up, false to take it down.</param>
77 /// <returns>The HTTP result (status codes described above).</returns>
78 [HttpPut("api/mock-services/{name}/availability")]
79 public IActionResult SetAvailability(string name, MockAvailabilityDto request)
80 {
81 var found = false;
82 if (_hub.Get(name) is MockExternalService mock) { mock.IsAvailable = request.Available; found = true; }
83 if (name.Equals(ExternalServiceNames.Staff, StringComparison.OrdinalIgnoreCase) && _staff is MockStaffDirectory staff) { staff.IsAvailable = request.Available; found = true; }
84 if (name.Equals(ExternalServiceNames.PublicTransport, StringComparison.OrdinalIgnoreCase) && _transit is MockTransitService transit) { transit.IsAvailable = request.Available; found = true; }
85 if (name.Equals(ExternalServiceNames.PropertyRegistration, StringComparison.OrdinalIgnoreCase) && _registry is MockPropertyRegistry registry) { registry.IsAvailable = request.Available; found = true; }
86 return found ? Ok(MockStates()) : NotFound(new { errors = new[] { $"No mock service named {name}." } });
87 }
88
89 private object MockStates() => new
90 {
91 outbound = _hub.All.OfType<MockExternalService>().Select(m => new { service = m.ServiceName, available = m.IsAvailable, received = m.Received }),
92 staffAvailable = (_staff as MockStaffDirectory)?.IsAvailable,
93 transitAvailable = (_transit as MockTransitService)?.IsAvailable,
94 propertyRegistryAvailable = (_registry as MockPropertyRegistry)?.IsAvailable
95 };
96}

26 /// <summary>
27 /// Routes and schedules from Public Transport (TRP-01), optionally only routes on a street.
28 /// Shows the source and update time; 503 with available = false when the data cannot be retrieved.
29 /// </summary>
30 /// <param name="street">Optional text the street name must contain.</param>
31 /// <returns>The HTTP result (status codes described above).</returns>
32 [HttpGet("routes")]
33 public IActionResult Routes([FromQuery] string? street)
34 {
35 try
36 {
37 var routes = _transit.GetRoutes(street);
38 return Ok(new
39 {
40 available = true,
41 source = ExternalServiceNames.PublicTransport,
42 retrievedAt = DateTime.UtcNow,
43 routes = routes.Select(r => new
44 {
45 r.RouteId,
46 r.Name,
47 r.Streets,
48 r.Schedule,
49 lastUpdated = r.LastUpdated,
50 lastUpdatedText = r.LastUpdated is null ? "Update time not supplied" : null
51 }),
52 message = routes.Count == 0 ? "No routes use this street." : null
53 });
54 }
55 catch (TransitUnavailableException ex)
56 {
57 return StatusCode(StatusCodes.Status503ServiceUnavailable, new
58 {
59 available = false,
60 source = ExternalServiceNames.PublicTransport,
61 message = $"Route and schedule data is unavailable: {ex.Message}"
62 });
63 }
64 }
65
66 /// <summary>Receives a route change / diversion (TRP-03). Requires X-Source-Service: Public Transport.</summary>
67 /// <param name="request">Route-change message sent by Public Transport.</param>
68 /// <returns>The HTTP result (status codes described above).</returns>
69 [HttpPost("diversions")]
70 public IActionResult Receive(RouteDiversionDto request)
71 {
72 var outcome = _diversions.Receive(request, RequestContext.SourceService(Request));
73 return this.ToResult(outcome, d => outcome.Duplicate
74 ? Ok(new { duplicate = true, diversion = d })
75 : StatusCode(StatusCodes.Status201Created, d));
76 }
77
78 /// <summary>Received route changes, optionally only those linked to an intervention (TRP-03).</summary>
79 /// <param name="interventionId">Intervention ID.</param>
80 /// <returns>The HTTP result (status codes described above).</returns>
81 [HttpGet("diversions")]
82 public IActionResult Diversions([FromQuery] int? interventionId) => Ok(_diversions.GetAll(interventionId));
83
84 /// <summary>Links a route change to an intervention (TRP-03).</summary>

90
91 /// <summary>Removes a link (TRP-03).</summary>
92 /// <param name="id">Route change ID.</param>
93 /// <param name="interventionId">Intervention ID.</param>
94 /// <returns>The HTTP result (status codes described above).</returns>
95 [HttpDelete("diversions/{id:int}/links/{interventionId:int}")]
96 public IActionResult Unlink(int id, int interventionId) => this.ToResult(_diversions.Unlink(id, interventionId));

26 /// <summary>
27 /// Receives a weather alert from E-services (ESV-03).
28 /// 201 with the new alert; 200 with { duplicate: true, alert } when it was already received; 400 invalid.
29 /// </summary>
30 /// <param name="request">Alert message sent by E-services.</param>
31 /// <returns>The HTTP result (status codes described above).</returns>
32 [HttpPost]
33 public IActionResult Receive(IncomingWeatherAlertDto request)
34 {
35 var outcome = _service.Receive(request);
36 return this.ToResult(outcome, alert => outcome.Duplicate
37 ? Ok(new { duplicate = true, alert })
38 : CreatedAtAction(nameof(GetById), new { id = alert.Id }, alert));
39 }

41 /// <summary>Lists alerts, newest first (ESV-03). Optional ?reviewStatus=Awaiting Review|Reviewed.</summary>
42 /// <param name="reviewStatus">Awaiting Review or Reviewed.</param>
43 /// <returns>The HTTP result (status codes described above).</returns>
44 [HttpGet]
45 public IActionResult GetAll([FromQuery] string? reviewStatus)
46 {
47 if (!string.IsNullOrWhiteSpace(reviewStatus)
48 && !WeatherAlertRules.ReviewStatuses.Contains(reviewStatus.Trim(), StringComparer.OrdinalIgnoreCase))
49 return BadRequest(new { errors = new[] { "Review status must be Awaiting Review or Reviewed." } });
50 return Ok(_service.GetAll(reviewStatus));
51 }

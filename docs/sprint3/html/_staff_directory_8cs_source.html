46 new StaffMember { PersonnelId = 103, Name = "Jordan Technician", Specializations = { "Electrical", "Streetlights" } },

49
50 /// <summary>When false, calls fail as if Staff were down.</summary>
51 public bool IsAvailable { get; set; } = true;
52
53 /// <inheritdoc />
54 public IReadOnlyList<StaffMember> GetPersonnel(string? specialization = null)
55 {
56 if (!IsAvailable) throw new StaffUnavailableException("The Staff service is currently unavailable.");
57
58 return _staff
59 .Where(s => string.IsNullOrWhiteSpace(specialization)
60 || s.Specializations.Contains(specialization.Trim(), StringComparer.OrdinalIgnoreCase))
61 .OrderBy(s => s.Name)
62 .ToList();
63 }

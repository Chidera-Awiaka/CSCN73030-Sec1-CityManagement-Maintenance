3/// <summary>
4/// One message exchanged with an external service (TEC-04). Every outbound send and
5/// every inbound message is recorded, so failed exchanges are visible and can be retried.
6/// </summary>
7public class ExternalExchange
8{
9 /// <summary>Exchange ID.</summary>
10 public int Id { get; set; }
11
12 /// <summary>External service, e.g. Budget.</summary>
13 public string Service { get; set; } = "";
14
15 /// <summary>Outbound (Maintenance sent it) or Inbound (Maintenance received it).</summary>
16 public string Direction { get; set; } = ExchangeRules.Outbound;
17
18 /// <summary>Operation from the contract, e.g. BudgetRequest, RoadClosureNotice.</summary>
19 public string Operation { get; set; } = "";
20
21 /// <summary>
22 /// Idempotency key. Outbound: built by Maintenance from the business data, so a
23 /// repeated send or retry does not create a second message. Inbound: the sender's
24 /// external ID, so a repeated delivery is detected.
25 /// </summary>
26 public string Key { get; set; } = "";
27
28 /// <summary>Intervention the exchange relates to, when there is one.</summary>
29 public int? InterventionId { get; set; }
30
31 /// <summary>Message body (JSON).</summary>
32 public string PayloadJson { get; set; } = "";
33
34 /// <summary>Succeeded, Failed or Received.</summary>
35 public string Status { get; set; } = ExchangeRules.Failed;
36
37 /// <summary>Number of send attempts (outbound).</summary>
38 public int Attempts { get; set; }
39
40 /// <summary>Last error message, when the last attempt failed.</summary>
41 public string? LastError { get; set; }
42
43 /// <summary>Reference returned by the external service on success.</summary>
44 public string? ExternalReference { get; set; }
45
46 /// <summary>When the exchange was first recorded (UTC).</summary>
47 public DateTime CreatedAt { get; set; } = DateTime.UtcNow;
48
49 /// <summary>When the last attempt was made (UTC).</summary>
50 public DateTime? LastAttemptAt { get; set; }
51}

53/// <summary>Allowed values for <see cref="ExternalExchange"/> (TEC-04).</summary>
54public static class ExchangeRules
55{
56 /// <summary>Message sent by Maintenance.</summary>
57 public const string Outbound = "Outbound";
58 /// <summary>Message received by Maintenance.</summary>
59 public const string Inbound = "Inbound";
60 /// <summary>Outbound message accepted by the external service.</summary>
61 public const string Succeeded = "Succeeded";
62 /// <summary>Outbound message not delivered; can be retried.</summary>
63 public const string Failed = "Failed";
64 /// <summary>Inbound message received and processed.</summary>
65 public const string Received = "Received";

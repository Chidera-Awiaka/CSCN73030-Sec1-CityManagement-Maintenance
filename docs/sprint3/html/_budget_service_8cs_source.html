20/// <summary>
21/// Budget integration.
22/// BUD-01 estimate (repair, material, equipment; automatic total),
23/// BUD-02 submit to Budget (intervention ID, breakdown, total; keeps the external reference; status Pending),
24/// BUD-03 receive the decision (linked to the request; details recorded; visible on the intervention),
25/// BUD-05 compare estimated and actual cost.
26/// BUD-04 is enforced through <see cref="BudgetPolicy"/>.
27/// </summary>
28public class BudgetService
29{

31 private readonly InterventionService _interventions;
32 private readonly ExchangeService _exchanges;

47 /// <summary>Enters or replaces the cost estimate (BUD-01). Not allowed while Pending or Approved.</summary>
48 /// <param name="id">Intervention ID.</param>
49 /// <param name="request">Repair, material and equipment costs.</param>
50 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
51 /// <returns>Success with the record, or the failure kind and messages.</returns>
52 public ServiceOutcome<Intervention> Estimate(int id, CostEstimateDto request, string actor)
53 {
54 if (request is null) return ServiceOutcome<Intervention>.Fail(OutcomeKind.Invalid, "Request body is required.");
55 var errors = new List<string>();
56 if (request.RepairCost < 0) errors.Add("Repair cost cannot be negative.");
57 if (request.MaterialCost < 0) errors.Add("Material cost cannot be negative.");
58 if (request.EquipmentCost < 0) errors.Add("Equipment cost cannot be negative.");
59 if (request.RepairCost + request.MaterialCost + request.EquipmentCost <= 0) errors.Add("The estimate total must be greater than zero.");
60 if (errors.Count > 0) return ServiceOutcome<Intervention>.Fail(OutcomeKind.Invalid, errors);
61
62 var intervention = _interventions.GetById(id);
63 if (intervention is null) return ServiceOutcome<Intervention>.Fail(OutcomeKind.NotFound, "Intervention not found.");
64 if (intervention.BudgetStatus is BudgetRules.Pending or BudgetRules.Approved)
65 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Conflict,
66 $"The estimate cannot be changed while the budget status is {intervention.BudgetStatus}.");
67
68 var estimate = new CostEstimate
69 {
70 RepairCost = request.RepairCost,
71 MaterialCost = request.MaterialCost,
72 EquipmentCost = request.EquipmentCost,
73 EstimatedBy = actor,
74 EstimatedAt = DateTime.UtcNow
75 };
76 _interventions.Update(id, i => i.CostEstimate = estimate);
77 _audit.Record("Intervention", id, AuditActions.CostEstimated, actor,
78 $"Repair {estimate.RepairCost:0.00} + material {estimate.MaterialCost:0.00} + equipment {estimate.EquipmentCost:0.00} = {estimate.Total:0.00}");
79 return ServiceOutcome<Intervention>.Ok(_interventions.GetById(id)!);
80 }

82 /// <summary>
83 /// Submits the estimate to Budget (BUD-02). On success the status becomes Pending and the
84 /// Budget reference is kept. Submitting the same estimate again does not create a duplicate
85 /// request (TEC-04); a failed submission can be retried.
86 /// </summary>
87 /// <param name="id">Intervention ID.</param>
88 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
89 /// <returns>Success with the record, or the failure kind and messages.</returns>
90 public ServiceOutcome<Intervention> Submit(int id, string actor)
91 {
92 var intervention = _interventions.GetById(id);
93 if (intervention is null) return ServiceOutcome<Intervention>.Fail(OutcomeKind.NotFound, "Intervention not found.");
94 if (intervention.CostEstimate is null)
95 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Invalid, "Enter a cost estimate before submitting it to Budget.");
96 if (intervention.BudgetStatus == BudgetRules.Approved)
97 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Conflict, "The budget is already approved.");
98 if (intervention.BudgetStatus == BudgetRules.Rejected
99 && intervention.BudgetDecision is not null
100 && intervention.CostEstimate.EstimatedAt <= intervention.BudgetDecision.ReceivedAt)
101 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Conflict,
102 "The previous estimate was rejected. Enter a revised estimate before submitting again.");
103
104 var e = intervention.CostEstimate;
105 var payload = JsonSerializer.Serialize(new
106 {
107 interventionId = id,
108 description = intervention.Description,
109 costBreakdown = new { repair = e.RepairCost, material = e.MaterialCost, equipment = e.EquipmentCost },
110 total = e.Total,
111 requestedBy = actor
112 });
113 var key = $"budget-request:{id}:{e.EstimatedAt:O}";
114 var sent = _exchanges.Send(ExternalServiceNames.Budget, Operation, key, payload, id);
115
116 if (!sent.Succeeded)
117 {
118 return new ServiceOutcome<Intervention>(OutcomeKind.ExternalFailure, null, new List<string>
119 {
120 $"The Budget service did not accept the request: {sent.Exchange.LastError} " +
121 $"It is recorded as exchange #{sent.Exchange.Id} and can be retried."
122 });
123 }
124
125 ApplySubmitted(id, sent.Exchange.ExternalReference!);
126 if (!sent.AlreadySent)
127 _audit.Record("Intervention", id, AuditActions.BudgetSubmitted, actor, $"Estimate {e.Total:0.00} submitted, reference {sent.Exchange.ExternalReference}");
128 return ServiceOutcome<Intervention>.Ok(_interventions.GetById(id)!, duplicate: sent.AlreadySent);
129 }

131 /// <summary>
132 /// Receives a decision from Budget (BUD-03), linked by the request reference. Only the Budget
133 /// service may send it. A repeated delivery is ignored (TEC-04).
134 /// </summary>
135 /// <param name="request">Decision message sent by Budget.</param>
136 /// <param name="sourceService">Value of the X-Source-Service header (the calling service).</param>
137 /// <returns>Success with the record, or the failure kind and messages.</returns>
138 public ServiceOutcome<Intervention> ReceiveDecision(BudgetDecisionDto request, string? sourceService)
139 {
140 if (!string.Equals(sourceService, ExternalServiceNames.Budget, StringComparison.OrdinalIgnoreCase))
141 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Forbidden, "Only the Budget service can send budget decisions (X-Source-Service: Budget).");
142
143 var errors = new List<string>();
144 if (request is null) return ServiceOutcome<Intervention>.Fail(OutcomeKind.Invalid, "Request body is required.");
145 if (string.IsNullOrWhiteSpace(request.DecisionId)) errors.Add("Decision ID is required.");
146 if (string.IsNullOrWhiteSpace(request.RequestReference)) errors.Add("Request reference is required.");
147 var decision = BudgetRules.Decisions.FirstOrDefault(d => d.Equals(request.Decision?.Trim(), StringComparison.OrdinalIgnoreCase));
148 if (decision is null) errors.Add("Decision must be Approved or Rejected.");
149 if (request.ApprovedAmount is < 0) errors.Add("Approved amount cannot be negative.");
150 if (errors.Count > 0) return ServiceOutcome<Intervention>.Fail(OutcomeKind.Invalid, errors);
151
152 var intervention = _interventions.GetAll().FirstOrDefault(i => i.BudgetRequestReference == request.RequestReference.Trim());
153 if (intervention is null)
154 return ServiceOutcome<Intervention>.Fail(OutcomeKind.NotFound, $"No budget request with reference {request.RequestReference}.");
155
156 var (_, duplicate) = _exchanges.RecordInbound(ExternalServiceNames.Budget, "BudgetDecision",
157 request.DecisionId.Trim(), JsonSerializer.Serialize(request), intervention.Id);
158 if (duplicate) return ServiceOutcome<Intervention>.Ok(intervention, duplicate: true);
159
160 if (intervention.BudgetStatus != BudgetRules.Pending)
161 return ServiceOutcome<Intervention>.Fail(OutcomeKind.Conflict,
162 $"Intervention {intervention.Id} is not waiting for a budget decision (status {intervention.BudgetStatus}).");
163
164 _interventions.Update(intervention.Id, i =>
165 {
166 i.BudgetStatus = decision!;
167 i.BudgetDecision = new BudgetDecision
168 {
169 Decision = decision!,
170 ApprovedAmount = request.ApprovedAmount,
171 Comments = request.Comments?.Trim(),
172 DecidedBy = request.DecidedBy?.Trim(),
173 DecisionId = request.DecisionId.Trim(),
174 ReceivedAt = DateTime.UtcNow
175 };
176 });
177 _audit.RecordFromService("Intervention", intervention.Id, AuditActions.BudgetDecisionReceived, ExternalServiceNames.Budget,
178 $"{decision}{(request.ApprovedAmount is null ? "" : $" ({request.ApprovedAmount:0.00})")}" +
179 $"{(string.IsNullOrWhiteSpace(request.DecidedBy) ? "" : $" by {request.DecidedBy}")}: {request.Comments}");
180 return ServiceOutcome<Intervention>.Ok(_interventions.GetById(intervention.Id)!);
181 }

183 /// <summary>Compares estimated and actual cost (BUD-05).</summary>
184 /// <param name="id">Intervention ID.</param>
185 /// <returns>Success with the record, or the failure kind and messages.</returns>
186 public ServiceOutcome<CostComparison> Compare(int id)
187 {
188 var i = _interventions.GetById(id);
189 if (i is null) return ServiceOutcome<CostComparison>.Fail(OutcomeKind.NotFound, "Intervention not found.");
190
191 var estimated = i.CostEstimate?.Total;
192 var actual = i.ActualCost;
193 if (estimated is null || actual is null)
194 {
195 var missing = estimated is null && actual is null ? "there is no estimate and no actual cost yet"
196 : estimated is null ? "there is no estimate yet" : "there is no actual cost yet (it is recorded when the intervention is closed)";
197 return ServiceOutcome<CostComparison>.Ok(new CostComparison(id, estimated, actual, null, null, false,
198 $"Comparison not available: {missing}."));
199 }
200
201 var difference = actual.Value - estimated.Value;
202 var percent = estimated.Value == 0 ? (decimal?)null : Math.Round(difference / estimated.Value * 100, 1);
203 return ServiceOutcome<CostComparison>.Ok(new CostComparison(id, estimated, actual, difference, percent, true, null));
204 }

206 private void ApplySubmitted(int id, string reference) =>
207 _interventions.Update(id, i =>
208 {
209 i.BudgetStatus = BudgetRules.Pending;
210 i.BudgetRequestReference = reference;
211 i.BudgetDecision = null;
212 });

214 private void OnExchangeSucceeded(ExternalExchange exchange)
215 {
216 if (exchange.Operation != Operation || exchange.InterventionId is null) return;
217 var i = _interventions.GetById(exchange.InterventionId.Value);
218 if (i is not null && i.BudgetStatus is BudgetRules.NotSubmitted or BudgetRules.Rejected
219 && i.BudgetRequestReference != exchange.ExternalReference)
220 {
221 ApplySubmitted(i.Id, exchange.ExternalReference!);
222 _audit.Record("Intervention", i.Id, AuditActions.BudgetSubmitted, "exchange retry",
223 $"Estimate {i.CostEstimate?.Total:0.00} submitted on retry (exchange #{exchange.Id}), reference {exchange.ExternalReference}");
224 }

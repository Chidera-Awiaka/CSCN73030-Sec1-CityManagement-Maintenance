71 /// <summary>
72 /// Returns interventions matching the supplied filters (MNT-02).
73 /// Any empty filter is ignored.
74 ///
75 /// Matching is case-insensitive.
76 /// Location matches when it contains the supplied text.
77 ///
78 /// Results are sorted by priority and then creation date.
79 /// </summary>
80 /// <param name="priority">Priority such as High.</param>
81 /// <param name="location">Part of the location.</param>
82 /// <param name="status">Operational status.</param>
83 /// <param name="technicianId">
84 /// Optional ID of the assigned technician (MNT-05 data).
85 /// Added in Sprint 2 to complete the MNT-02 filter acceptance
86 /// criteria (GitHub #2). Ignored when null.
87 /// </param>
88 public List<Intervention> Filter(
89 string? priority,
90 string? location,
91 string? status,
92 int? technicianId = null)
93 {
94 lock (_lock)
95 {
96 IEnumerable<Intervention> query = _interventions;
97
98 if (!string.IsNullOrWhiteSpace(priority))
99 {
100 query = query.Where(x =>
101 string.Equals(
102 x.Priority,
103 priority.Trim(),
104 StringComparison.OrdinalIgnoreCase));
105 }
106
107 if (!string.IsNullOrWhiteSpace(location))
108 {
109 query = query.Where(x =>
110 x.Location.Contains(
111 location.Trim(),
112 StringComparison.OrdinalIgnoreCase));
113 }
114
115 if (!string.IsNullOrWhiteSpace(status))
116 {
117 query = query.Where(x =>
118 string.Equals(
119 x.Status,
120 status.Trim(),
121 StringComparison.OrdinalIgnoreCase));
122 }
123
124 if (technicianId is not null)
125 {
126 query = query.Where(x =>
127 x.AssignedTechnicianId == technicianId);
128 }
129
130 return query
131 .OrderByDescending(x =>
132 Array.IndexOf(
133 InterventionRules.Priorities,
134 x.Priority))
135 .ThenByDescending(x => x.CreatedAt)
136 .ToList();
137 }
138 }

185 /// <summary>
186 /// Creates a new intervention (MNT-01).
187 /// A unique ID is generated.
188 /// Initial status is always To Do.
189 /// </summary>
190 /// <param name="request">
191 /// Request containing intervention information.
192 /// </param>
193 /// <returns>The created intervention.</returns>
194 /// <exception cref="ArgumentException">
195 /// Thrown when validation fails.
196 /// </exception>
197 public Intervention Create(
198 CreateInterventionDto request)
199 {
200 var errors = Validate(request);
201
202 if (errors.Count > 0)
203 {
204 throw new ArgumentException(
205 string.Join(" ", errors));
206 }
207
208 lock (_lock)
209 {
210 var intervention = new Intervention
211 {
212 Id = _interventions.Count == 0
213 ? 1
214 : _interventions.Max(x => x.Id) + 1,
215
216 Description = request.Description.Trim(),
217
218 Location = request.Location.Trim(),
219
220 IssueType = request.IssueType.Trim(),
221
222 Priority =
223 string.IsNullOrWhiteSpace(request.Priority)
224 ? InterventionRules.DefaultPriority
225 : request.Priority.Trim(),
226
227 Status = InterventionRules.InitialStatus,
228
229 CreatedAt = DateTime.UtcNow,
230
231 Instructions =
232 string.IsNullOrWhiteSpace(request.Instructions)
233 ? null
234 : request.Instructions.Trim()
235 };
236
237 _interventions.Add(intervention);
238
239 Save();
240
241 return intervention;
242 }
243 }

441 /// <summary>Removes a property link from an intervention (REG-02).</summary>
442 /// <param name="id">Intervention ID.</param>
443 /// <param name="propertyId">External Property ID to remove.</param>
444 /// <returns>Success, InterventionNotFound or NotLinked.</returns>
445 public PropertyLinkResult UnlinkProperty(int id, string propertyId)
446 {
447 lock (_lock)
448 {
449 var intervention = _interventions.FirstOrDefault(x => x.Id == id);
450 if (intervention == null)

452 return PropertyLinkResult.InterventionNotFound;

490 /// <summary>
491 /// Checks a close request against the MNT-08 rules.
492 /// </summary>
493 /// <param name="request">Close request.</param>
494 /// <param name="createdAt">When the intervention was created.</param>
495 /// <param name="now">Current time (UTC); a parameter so tests can fix it.</param>
496 /// <returns>List of validation errors; empty when valid.</returns>
497 public static List<string> ValidateClose(CloseInterventionDto? request, DateTime createdAt, DateTime now)
498 {
499 var errors = new List<string>();
500
501 if (request is null)
502 {
503 errors.Add("Request body is required.");
504 return errors;
505 }
506
507 if (string.IsNullOrWhiteSpace(request.CompletionReport))
508 {
509 errors.Add("A completion report is required.");
510 }
511
512 if (request.ActualCompletionDate is { } done)
513 {
514 if (done > now.AddMinutes(5))
515 {
516 errors.Add("Actual completion date cannot be in the future.");
517 }
518
519 if (done < createdAt)
520 {
521 errors.Add("Actual completion date cannot be before the intervention was created.");
522 }
523 }
524
525 if (request.ActualCost is < 0)
526 {
527 errors.Add("Actual cost cannot be negative.");
528 }
529
530 return errors;
531 }

533 /// <summary>
534 /// Closes an intervention (MNT-08): records the completion report, the actual
535 /// completion date (now when not supplied) and the optional actual cost, and sets
536 /// the status to Completed. Call <see cref="ValidateClose"/> first.
537 /// </summary>
538 /// <param name="id">Intervention ID.</param>
539 /// <param name="request">Validated close request.</param>
540 /// <returns>Closed, NotFound or AlreadyClosed.</returns>
541 public CloseResult CloseIntervention(int id, CloseInterventionDto request)
542 {
543 lock (_lock)
544 {
545 var intervention = _interventions.FirstOrDefault(x => x.Id == id);
546 if (intervention == null)
547 {
548 return CloseResult.NotFound;
549 }
550
551 if (intervention.Status == InterventionRules.CompletedStatus
552 || intervention.Status == InterventionRules.CancelledStatus)
553 {
554 return CloseResult.AlreadyClosed;
555 }
556
557 intervention.CompletionReport = request.CompletionReport.Trim();
558 intervention.ActualCompletionDate = request.ActualCompletionDate ?? DateTime.UtcNow;
559 intervention.ActualCost = request.ActualCost;
560 intervention.Status = InterventionRules.CompletedStatus;
561
562 Save();
563 return CloseResult.Closed;
564 }
565 }

16/// <summary>
17/// Integration boundary with the Property Registration service of the
18/// City Management Enterprise System. Related requirements: REG-01, REG-02.
19/// </summary>
20/// <remarks>
21/// Sprint 2 uses <see cref="MockPropertyRegistry"/> (allowed by REG-01 while the
22/// Property Registration service is not available). When that service publishes its
23/// REST API, a client implementing this interface replaces the mock in Program.cs;
24/// the controllers and tests do not change.
25/// </remarks>
26public interface IPropertyRegistry
27{
28 /// <summary>
29 /// Searches properties by Property ID and/or address (REG-01).
30 /// </summary>
31 /// <param name="propertyId">Exact Property ID (case-insensitive). Ignored when empty.</param>
32 /// <param name="address">Text the address must contain (case-insensitive). Ignored when empty.</param>
33 /// <returns>Matching properties; an empty list when nothing matches.</returns>
34 /// <exception cref="PropertyRegistryUnavailableException">The registry cannot be reached.</exception>
35 IReadOnlyList<Property> Search(string? propertyId, string? address);
36
37 /// <summary>Gets one property by its Property ID (REG-02 validation).</summary>
38 /// <param name="propertyId">Property ID to look up.</param>
39 /// <returns>The property, or null when it does not exist.</returns>
40 /// <exception cref="PropertyRegistryUnavailableException">The registry cannot be reached.</exception>
41 Property? GetById(string propertyId);
42
43 /// <summary>
44 /// Gets the utility information supplied for a property (REG-03). A utility the
45 /// registry does not supply is returned as null so the UI can say so explicitly.
46 /// </summary>
47 /// <param name="propertyId">Property ID.</param>
48 /// <returns>The utilities, or null when the property does not exist.</returns>
49 /// <exception cref="PropertyRegistryUnavailableException">The registry cannot be reached.</exception>
50 PropertyUtilities? GetUtilities(string propertyId);
51}

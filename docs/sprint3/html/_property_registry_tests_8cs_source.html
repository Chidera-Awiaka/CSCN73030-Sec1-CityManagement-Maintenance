5/// <summary>
6/// Unit tests for the Property Registry search (REG-01).
7/// Test cases map to TC-REG-01 in the Release Note.
8/// </summary>
9public sealed class PropertyRegistryTests
10{
11 private readonly MockPropertyRegistry _registry = new();
12
13 /// <summary>Test: Search : By Property Id : Returns That Property.</summary>
14[Fact]
15 public void Search_ByPropertyId_ReturnsThatProperty()
16 {
17 var results = _registry.Search("PR-1003", null);
18
19 var property = Assert.Single(results);
20 Assert.Equal("PR-1003", property.PropertyId);
21 Assert.Equal("45 Victoria Street, Kitchener", property.Address);
22 }
23
24 /// <summary>Test: Search : By Property Id : Is Case Insensitive And Trimmed.</summary>
25[Fact]
26 public void Search_ByPropertyId_IsCaseInsensitiveAndTrimmed()
27 {
28 var results = _registry.Search(" pr-1001 ", null);
29
30 Assert.Equal("PR-1001", Assert.Single(results).PropertyId);
31 }
32
33 /// <summary>Test: Search : By Address Text : Returns All Matching.</summary>
34[Fact]
35 public void Search_ByAddressText_ReturnsAllMatching()
36 {
37 var results = _registry.Search(null, "king street");
38
39 Assert.Equal(2, results.Count);
40 Assert.All(results, p => Assert.Contains("King Street", p.Address));
41 }
42
43 /// <summary>Test: Search : By Id And Address : Must Match Both.</summary>
44[Fact]
45 public void Search_ByIdAndAddress_MustMatchBoth()
46 {
47 Assert.Single(_registry.Search("PR-1001", "King"));
48 Assert.Empty(_registry.Search("PR-1001", "Victoria"));
49 }
50
51 /// <summary>Test: Search : No Match : Returns Empty List.</summary>
52[Fact]
53 public void Search_NoMatch_ReturnsEmptyList()
54 {
55 Assert.Empty(_registry.Search(null, "Nowhere Road"));
56 Assert.Empty(_registry.Search("PR-9999", null));
57 }
58
59 /// <summary>Test: Get By Id : Unknown Or Blank : Returns Null.</summary>
60[Fact]
61 public void GetById_UnknownOrBlank_ReturnsNull()
62 {
63 Assert.Null(_registry.GetById("PR-9999"));
64 Assert.Null(_registry.GetById(" "));
65 }
66
67 /// <summary>Test: Search : Registry Unavailable : Throws.</summary>
68[Fact]
69 public void Search_RegistryUnavailable_Throws()
70 {
71 _registry.IsAvailable = false;
72
73 Assert.Throws<PropertyRegistryUnavailableException>(() => _registry.Search("PR-1001", null));
74 Assert.Throws<PropertyRegistryUnavailableException>(() => _registry.GetById("PR-1001"));

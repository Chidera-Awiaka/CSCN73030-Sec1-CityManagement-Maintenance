120 private static InterventionService ServiceWithData()

138 [Fact]
139 public void Filter_ByPriority_ReturnsOnlyMatching()

156 [Fact]
157 public void Filter_ByStatus_ReturnsOnlyMatching()
158 {

161 service.UpdateStatus(target.Id, "On Hold");
162

164

187 public void Filter_SortsByPriorityHighestFirst()

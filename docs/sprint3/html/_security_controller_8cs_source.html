21 /// <summary>
22 /// Sends an assistance request to Security (SEC-01). 201 with the request and its transmission
23 /// status (Succeeded or Failed; a failed send can be retried via /api/exchanges/{exchangeId}/retry).
24 /// </summary>
25 /// <param name="id">Intervention ID.</param>
26 /// <param name="request">Reason, location and period.</param>

32 /// <summary>Requests for one intervention, with transmission status and any response (SEC-01, SEC-02).</summary>
33 /// <param name="id">Intervention ID.</param>
34 /// <returns>The HTTP result (status codes described above).</returns>
35 [HttpGet("api/interventions/{id:int}/security-requests")]
36 public IActionResult ForIntervention(int id) => Ok(_service.GetAll(id));
37
38 /// <summary>All requests.</summary>
39 /// <returns>The HTTP result (status codes described above).</returns>
40 [HttpGet("api/security/requests")]
41 public IActionResult All() => Ok(_service.GetAll());

43 /// <summary>
44 /// Receives Security's response (SEC-02). Requires header X-Source-Service: Security; Maintenance
45 /// users get 403 because they cannot approve requests on Security's behalf.
46 /// </summary>
47 /// <param name="response">Response message from Security.</param>
48 /// <returns>The HTTP result (status codes described above).</returns>
49 [HttpPost("api/security/responses")]
50 public IActionResult Respond(SecurityResponseDto response)
51 {
52 var outcome = _service.ReceiveResponse(response, RequestContext.SourceService(Request));
53 return this.ToResult(outcome, r => Ok(new { duplicate = outcome.Duplicate, request = r }));
54 }

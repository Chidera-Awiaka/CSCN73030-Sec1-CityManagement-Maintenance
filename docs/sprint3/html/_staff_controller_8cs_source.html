23 /// <summary>
24 /// Personnel with IDs, names and specializations from Staff, optionally filtered by

26 /// </summary>
27 /// <param name="specialization">Optional specialization filter, e.g. Road.</param>

30 public IActionResult Personnel([FromQuery] string? specialization)
31 {
32 try
33 {
34 var people = _staff.GetPersonnel(specialization);
35 return Ok(new
36 {
37 source = ExternalServiceNames.Staff,
38 retrievedAt = DateTime.UtcNow,
39 specializations = _staff.GetPersonnel().SelectMany(p => p.Specializations).Distinct().OrderBy(s => s),
40 personnel = people
41 });
42 }
43 catch (StaffUnavailableException ex)
44 {
45 return StatusCode(StatusCodes.Status503ServiceUnavailable, new { errors = new[] { ex.Message } });
46 }
47 }

49 /// <summary>Overlapping assignments for an intervention's technician and planned period (STF-02).</summary>
50 /// <param name="id">Intervention ID.</param>
51 /// <returns>The HTTP result (status codes described above).</returns>
52 [HttpGet("api/interventions/{id:int}/conflicts")]
53 public IActionResult Conflicts(int id)
54 {
55 var target = _interventions.GetById(id);
56 if (target is null) return NotFound(new { errors = new[] { "Intervention not found." } });
57 var conflicts = AssignmentConflicts.For(_interventions.GetAll(), target);
58 return Ok(new { interventionId = id, hasConflicts = conflicts.Count > 0, conflicts, scope = AssignmentConflicts.Scope });
59 }

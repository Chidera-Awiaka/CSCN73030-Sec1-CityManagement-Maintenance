8/// <summary>

18public class IncidentReportsController : ControllerBase

22 /// <summary>Creates the controller.</summary>

28 /// <summary>
29 /// Receives an incident report from E-services (ESV-01).
30 /// Returns 201 with the stored report, 400 with validation errors,
31 /// or 409 when the same E-services report was already received.
32 /// </summary>
33 [HttpPost]
34 public IActionResult Receive(IncomingIncidentReportDto request)
35 {
36 var outcome = _service.Receive(request);
37 return outcome.Result == ReviewResult.Success
38 ? CreatedAtAction(nameof(GetById), new { id = outcome.Report!.Id }, outcome.Report)
39 : ToError(outcome);
40 }
41
42 /// <summary>
43 /// Lists incident reports, newest first (ESV-01).
44 /// Example: GET /api/incident-reports?status=Pending
45 /// </summary>
46 /// <param name="status">Optional review state: Pending, Accepted or Rejected.</param>
47 [HttpGet]
48 public IActionResult GetAll([FromQuery] string? status)
49 {
50 if (!string.IsNullOrWhiteSpace(status)
51 && !IncidentReportRules.Statuses.Contains(status.Trim(), StringComparer.OrdinalIgnoreCase))
52 {
53 return BadRequest(new { errors = new[] { "Status must be Pending, Accepted or Rejected." } });
54 }
55
56 return Ok(_service.GetAll(status));
57 }
58
59 /// <summary>Gets one incident report by ID.</summary>
60 [HttpGet("{id:int}")]
61 public IActionResult GetById(int id)
62 {
63 var report = _service.GetById(id);
64 return report is null ? NotFound(new { errors = new[] { "Incident report not found." } }) : Ok(report);
65 }
66
67 /// <summary>
68 /// Accepts a pending report and creates an intervention from it (ESV-02).
69 /// Returns 200, 400, 404, or 409 when it was already reviewed.
70 /// </summary>
71 [HttpPost("{id:int}/accept")]
72 public IActionResult Accept(int id, AcceptIncidentReportDto request)
73 {
74 var outcome = _service.Accept(id, request);
75 return outcome.Result == ReviewResult.Success ? Ok(outcome.Report) : ToError(outcome);
76 }
77
78 /// <summary>
79 /// Rejects a pending report with a reason (ESV-02).
80 /// Returns 200, 400, 404, or 409 when it was already reviewed.
81 /// </summary>
82 [HttpPost("{id:int}/reject")]
83 public IActionResult Reject(int id, RejectIncidentReportDto request)
84 {
85 var outcome = _service.Reject(id, request);
86 return outcome.Result == ReviewResult.Success ? Ok(outcome.Report) : ToError(outcome);
87 }
88
89 /// <summary>Maps a failed outcome to the matching HTTP status code.</summary>
90 private IActionResult ToError(ReviewOutcome outcome)
91 {
92 var body = new { errors = outcome.Errors };
93 return outcome.Result switch
94 {
95 ReviewResult.NotFound => NotFound(body),
96 ReviewResult.Conflict => Conflict(body),
97 _ => BadRequest(body)
98 };
99 }
100}

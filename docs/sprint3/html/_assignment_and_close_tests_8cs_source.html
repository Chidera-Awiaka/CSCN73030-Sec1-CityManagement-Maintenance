157 /// <summary>Test: completion date cannot be in the future or before creation; cost cannot be negative.</summary>
158 [Fact]
159 public void ValidateClose_BadDatesAndNegativeCost_Errors()
160 {
161 var now = new DateTime(2026, 11, 12, 12, 0, 0, DateTimeKind.Utc);
162 var created = now.AddDays(-2);
163
164 var future = InterventionService.ValidateClose(new CloseInterventionDto { CompletionReport = "x", ActualCompletionDate = now.AddDays(1) }, created, now);
165 var early = InterventionService.ValidateClose(new CloseInterventionDto { CompletionReport = "x", ActualCompletionDate = created.AddDays(-1) }, created, now);
166 var negative = InterventionService.ValidateClose(new CloseInterventionDto { CompletionReport = "x", ActualCost = -1 }, created, now);
167 var valid = InterventionService.ValidateClose(new CloseInterventionDto { CompletionReport = "x", ActualCompletionDate = now.AddHours(-1), ActualCost = 0 }, created, now);
168
169 Assert.Contains("Actual completion date cannot be in the future.", future);
170 Assert.Contains("Actual completion date cannot be before the intervention was created.", early);
171 Assert.Contains("Actual cost cannot be negative.", negative);
172 Assert.Empty(valid);
173 }

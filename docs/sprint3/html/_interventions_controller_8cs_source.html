135 /// <summary>
136 /// Updates the priority of an intervention (MNT-03).
137 /// </summary>
138 /// <param name="id">Intervention ID.</param>
139 /// <param name="request">New priority: Low, Medium, High or Critical.</param>
140 /// <returns>The HTTP result (status codes described above).</returns>
141 [HttpPut("{id:int}/priority")]
142 public IActionResult UpdatePriority(
143 int id,
144 UpdatePriorityDto request)
145 {
146 if (string.IsNullOrWhiteSpace(request.Priority))
147 {
148 return BadRequest("Priority is required.");
149 }
150
151 var priority = request.Priority.Trim();
152
153 if (!InterventionRules.Priorities.Contains(priority))
154 {
155 return BadRequest(
156 "Invalid priority. Allowed values: Low, Medium, High, Critical."
157 );
158 }
159
160 var previousPriority = _service.GetById(id)?.Priority;
161 if (!_service.UpdatePriority(id, priority))
162 {
163 return NotFound("Intervention not found.");
164 }
165
166 Audit(id, AuditActions.PriorityChanged, $"{previousPriority} -> {priority}");
167
168 return Ok(new
169 {
170 message = "Priority updated successfully."
171 });
172 }

174 /// <summary>
175 /// Updates the operational status of an intervention (MNT-07).
176 /// </summary>
177 /// <param name="id">Intervention ID.</param>
178 /// <param name="request">New operational status.</param>
179 /// <returns>The HTTP result (status codes described above).</returns>
180 [HttpPut("{id:int}/status")]
181 public IActionResult UpdateStatus(
182 int id,
183 UpdateStatusDto request)
184 {
185 if (string.IsNullOrWhiteSpace(request.Status))
186 {
187 return BadRequest("Status is required.");
188 }
189
190 var status = request.Status.Trim();
191
192 if (!InterventionRules.Statuses.Contains(status))
193 {
194 return BadRequest(
195 "Invalid status."
196 );
197 }
198
199 var current = _service.GetById(id);
200 if (current is null)
201 {
202 return NotFound("Intervention not found.");
203 }
204
205 // BUD-04: work cannot start without an approved budget (when the rule is enabled).
206 if (_budgetPolicy.BlockReason(current, status) is { } blocked)
207 {
208 return Conflict(new { message = blocked, errors = new[] { blocked } });
209 }
210
211 var previousStatus = current.Status;
212 if (!_service.UpdateStatus(id, status))
213 {
214 return NotFound("Intervention not found.");
215 }
216
217 Audit(id, AuditActions.StatusChanged, $"{previousStatus} -> {status}");
218
219 return Ok(new

221 message = "Status updated successfully."
222 });

225 /// <summary>
226 /// Schedules an intervention (MNT-04).
227 ///
228 /// Stores planned start and end dates.
229 /// The planned end date cannot be earlier than
230 /// the planned start date.
231 ///
232 /// A successfully scheduled intervention is
233 /// moved to the Scheduled operational status.
234 /// </summary>
235 /// <param name="id">Intervention ID.</param>
236 /// <param name="request">Planned start and end dates.</param>
237 /// <returns>The HTTP result (status codes described above).</returns>
238 [HttpPut("{id:int}/schedule")]
239 public IActionResult ScheduleIntervention(
240 int id,
241 ScheduleInterventionDto request)
242 {
243 if (request.PlannedStartDate == default)
244 {
245 return BadRequest(
246 "Planned start date is required."
247 );
248 }
249
250 if (request.PlannedEndDate == default)
251 {
252 return BadRequest(
253 "Planned end date is required."
254 );
255 }
256
257 if (request.PlannedEndDate < request.PlannedStartDate)
258 {
259 return BadRequest(
260 "Planned end date cannot be before planned start date."
261 );
262 }
263
264 bool updated = _service.ScheduleIntervention(
265 id,
266 request.PlannedStartDate,
267 request.PlannedEndDate
268 );
269
270 if (!updated)
271 {
272 return NotFound("Intervention not found.");
273 }
274
275 Audit(id, AuditActions.Scheduled, $"{request.PlannedStartDate:yyyy-MM-dd HH:mm} - {request.PlannedEndDate:yyyy-MM-dd HH:mm}");
276
277 return Ok(new
278 {
279 message = "Intervention scheduled successfully.",
280 overlaps = Overlaps(id)
281 });
282 }

333 /// <summary>
334 /// Lists the full details of the properties linked to an intervention (REG-02).
335 /// Details are read from the Property Registry using the stored Property IDs.
336 /// </summary>
337 /// <param name="id">Intervention ID.</param>
338 /// <returns>200 with the property details; 404; 503 when the registry is unavailable.</returns>
339 [HttpGet("{id:int}/properties")]
340 public IActionResult GetLinkedProperties(int id)
341 {
342 var intervention = _service.GetById(id);
343 if (intervention is null)
344 {
345 return NotFound(new { message = "Intervention not found." });
346 }
347
348 try
349 {
350 var properties = intervention.LinkedPropertyIds
351 .Select(pid => _registry.GetById(pid)
352 ?? new Property { PropertyId = pid, Address = "(not found in registry)" })
353 .ToList();
354 return Ok(properties);
355 }
356 catch (PropertyRegistryUnavailableException ex)
357 {
358 return StatusCode(StatusCodes.Status503ServiceUnavailable,
359 new { message = ex.Message, linkedPropertyIds = intervention.LinkedPropertyIds });
360 }
361 }

363 /// <summary>
364 /// Links a property from the Property Registry to an intervention (REG-02).
365 /// </summary>
366 /// <param name="id">Intervention ID.</param>
367 /// <param name="request">The Property ID to link.</param>
368 /// <returns>
369 /// 200 when linked; 400 when no Property ID is given; 404 when the intervention or
370 /// property does not exist; 409 when already linked; 503 when the registry is unavailable.
371 /// </returns>
372 [HttpPost("{id:int}/properties")]
373 public IActionResult LinkProperty(int id, LinkPropertyDto request)
374 {
375 if (string.IsNullOrWhiteSpace(request.PropertyId))
376 {
377 return BadRequest(new { message = "Property ID is required." });
378 }
379
380 Property? property;
381 try
382 {
383 property = _registry.GetById(request.PropertyId);
384 }
385 catch (PropertyRegistryUnavailableException ex)
386 {
387 return StatusCode(StatusCodes.Status503ServiceUnavailable, new { message = ex.Message });
388 }
389
390 if (property is null)
391 {
392 return NotFound(new { message = $"Property '{request.PropertyId}' was not found in the Property Registry." });
393 }
394
395 return _service.LinkProperty(id, property.PropertyId) switch
396 {
397 PropertyLinkResult.Success => Ok(new { message = $"Property {property.PropertyId} linked to intervention {id}." }),
398 PropertyLinkResult.AlreadyLinked => Conflict(new { message = $"Property {property.PropertyId} is already linked." }),
399 _ => NotFound(new { message = "Intervention not found." })
400 };
401 }

437 /// <summary>
438 /// Closes an intervention (MNT-08). A completion report is required; the actual
439 /// completion date is recorded (now when omitted); the actual cost can be entered.
440 /// </summary>
441 /// <param name="id">Intervention ID.</param>
442 /// <param name="request">Completion report, optional completion date and cost.</param>
443 /// <returns>200 when closed; 400 with errors; 404 not found; 409 already Completed or Cancelled.</returns>
444 [HttpPut("{id:int}/close")]
445 public IActionResult CloseIntervention(int id, CloseInterventionDto request)
446 {
447 var intervention = _service.GetById(id);
448 if (intervention is null)
449 {
450 return NotFound(new { message = "Intervention not found." });
451 }
452
453 var errors = InterventionService.ValidateClose(request, intervention.CreatedAt, DateTime.UtcNow);
454 if (errors.Count > 0)
455 {
456 return BadRequest(new { errors });
457 }
458
459 var previousStatus = intervention.Status;
460 return _service.CloseIntervention(id, request) switch
461 {
462 CloseResult.Closed => AuditedClose(id, previousStatus),
463 CloseResult.AlreadyClosed => Conflict(new { message = $"Intervention {id} is already {intervention.Status}." }),
464 _ => NotFound(new { message = "Intervention not found." })
465 };
466 }
467
468 private IActionResult AuditedClose(int id, string previousStatus)
469 {
470 Audit(id, AuditActions.Closed, $"{previousStatus} -> Completed");
471 Audit(id, AuditActions.StatusChanged, $"{previousStatus} -> Completed");
472 return Ok(new { message = $"Intervention {id} closed." });
473 }

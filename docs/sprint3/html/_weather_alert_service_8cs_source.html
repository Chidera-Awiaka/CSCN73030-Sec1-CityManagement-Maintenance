63 /// <summary>Checks an incoming alert against the ESV-03 contract.</summary>
64 /// <param name="request">Alert message sent by E-services.</param>
65 /// <returns>The result.</returns>
66 public static List<string> Validate(IncomingWeatherAlertDto? request)
67 {
68 var errors = new List<string>();
69 if (request is null) { errors.Add("Request body is required."); return errors; }
70 if (string.IsNullOrWhiteSpace(request.ExternalAlertId)) errors.Add("External alert ID is required.");
71 if (string.IsNullOrWhiteSpace(request.Title)) errors.Add("Title is required.");
72 if (!string.IsNullOrWhiteSpace(request.Severity)
73 && !WeatherAlertRules.Severities.Contains(request.Severity.Trim(), StringComparer.OrdinalIgnoreCase))
74 errors.Add("Severity must be Minor, Moderate, Severe or Extreme.");
75 if (request.ValidFrom is not null && request.ValidTo is not null && request.ValidTo < request.ValidFrom)
76 errors.Add("Validity end cannot be before validity start.");
77 return errors;
78 }

80 /// <summary>
81 /// Receives an alert (ESV-03). A repeated delivery of the same E-services alert ID returns the
82 /// existing alert with Duplicate = true and creates nothing new (TEC-04).
83 /// </summary>
84 /// <param name="request">Alert message sent by E-services.</param>
85 /// <returns>Success with the record, or the failure kind and messages.</returns>
86 public ServiceOutcome<WeatherAlert> Receive(IncomingWeatherAlertDto request)
87 {
88 var errors = Validate(request);
89 if (errors.Count > 0) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Invalid, errors);
90
91 var externalId = request.ExternalAlertId.Trim();
92 lock (_store.Sync)
93 {
94 var existing = _store.Items.FirstOrDefault(a => a.ExternalAlertId == externalId);
95 if (existing is not null) return ServiceOutcome<WeatherAlert>.Ok(existing, duplicate: true);
96
97 var severity = string.IsNullOrWhiteSpace(request.Severity)
98 ? null
99 : WeatherAlertRules.Severities.First(s => s.Equals(request.Severity.Trim(), StringComparison.OrdinalIgnoreCase));
100
101 var alert = new WeatherAlert
102 {
103 Id = _store.NextId(a => a.Id),
104 ExternalAlertId = externalId,
105 Title = request.Title.Trim(),
106 Content = string.IsNullOrWhiteSpace(request.Content) ? request.Title.Trim() : request.Content.Trim(),
107 AffectedArea = string.IsNullOrWhiteSpace(request.AffectedArea) ? null : request.AffectedArea.Trim(),
108 Severity = severity,
109 ValidFrom = request.ValidFrom,
110 ValidTo = request.ValidTo,
111 ReceivedAt = DateTime.UtcNow,
112 ReviewStatus = WeatherAlertRules.AwaitingReview
113 };
114 _store.Items.Add(alert);
115 _store.Save();
116 _exchanges.RecordInbound(ExternalServiceNames.EServices, "WeatherAlert", externalId, JsonSerializer.Serialize(request));
117 return ServiceOutcome<WeatherAlert>.Ok(alert);
118 }
119 }

121 /// <summary>Records an assessment and marks the alert Reviewed (ESV-04).</summary>
122 /// <param name="id">Weather alert ID.</param>
123 /// <param name="request">Decision, rationale and optional reviewer.</param>
124 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
125 /// <returns>Success with the record, or the failure kind and messages.</returns>
126 public ServiceOutcome<WeatherAlert> Assess(int id, WeatherAssessmentDto request, string actor)
127 {
128 var reviewer = string.IsNullOrWhiteSpace(request?.Reviewer) ? actor : request!.Reviewer!.Trim();
129 var errors = new List<string>();
130 if (request is null) errors.Add("Request body is required.");
131 else
132 {
133 if (!WeatherAlertRules.Decisions.Contains(request.Decision?.Trim() ?? "", StringComparer.OrdinalIgnoreCase))
134 errors.Add("Decision must be Action Required, Monitor or No Impact.");
135 if (string.IsNullOrWhiteSpace(request.Rationale)) errors.Add("Rationale is required.");
136 if (reviewer == RequestContext.UnknownActor) errors.Add("Reviewer is required.");
137 }
138 if (errors.Count > 0) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Invalid, errors);
139
140 lock (_store.Sync)
141 {
142 var alert = _store.Items.FirstOrDefault(a => a.Id == id);
143 if (alert is null) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.NotFound, "Weather alert not found.");
144
145 var decision = WeatherAlertRules.Decisions.First(d => d.Equals((request!.Decision ?? "").Trim(), StringComparison.OrdinalIgnoreCase));
146 alert.Assessments.Add(new WeatherAssessment
147 {
148 Reviewer = reviewer,
149 ReviewedAt = DateTime.UtcNow,
150 Decision = decision,
151 Rationale = request!.Rationale.Trim()
152 });
153 alert.ReviewStatus = WeatherAlertRules.Reviewed;
154 _store.Save();
155 _audit.Record("WeatherAlert", alert.Id, AuditActions.WeatherAssessment, reviewer, $"{decision}: {request.Rationale.Trim()}");
156 return ServiceOutcome<WeatherAlert>.Ok(alert);
157 }
158 }

160 /// <summary>Sets the review state back to Awaiting Review (or Reviewed when an assessment exists) (ESV-04).</summary>
161 /// <param name="id">Weather alert ID.</param>
162 /// <param name="reviewStatus">Awaiting Review or Reviewed.</param>
163 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
164 /// <returns>Success with the record, or the failure kind and messages.</returns>
165 public ServiceOutcome<WeatherAlert> SetReviewStatus(int id, string reviewStatus, string actor)
166 {
167 var status = WeatherAlertRules.ReviewStatuses.FirstOrDefault(s => s.Equals(reviewStatus?.Trim(), StringComparison.OrdinalIgnoreCase));
168 if (status is null) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Invalid, "Review status must be Awaiting Review or Reviewed.");
169
170 lock (_store.Sync)
171 {
172 var alert = _store.Items.FirstOrDefault(a => a.Id == id);
173 if (alert is null) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.NotFound, "Weather alert not found.");
174 if (status == WeatherAlertRules.Reviewed && alert.Assessments.Count == 0)
175 return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Conflict, "Record an assessment before marking the alert Reviewed.");
176
177 var previous = alert.ReviewStatus;
178 alert.ReviewStatus = status;
179 _store.Save();
180 _audit.Record("WeatherAlert", alert.Id, AuditActions.WeatherAssessment, actor, $"Review status {previous} -> {status}");
181 return ServiceOutcome<WeatherAlert>.Ok(alert);
182 }
183 }

185 /// <summary>Takes an action from the alert (ESV-05). The decision is kept on the alert.</summary>
186 /// <param name="id">Weather alert ID.</param>
187 /// <param name="request">Action type and its details.</param>
188 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
189 /// <returns>Success with the record, or the failure kind and messages.</returns>
190 public ServiceOutcome<WeatherAlert> TakeAction(int id, WeatherAlertActionDto request, string actor)
191 {
192 if (request is null) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Invalid, "Request body is required.");
193 var type = WeatherAlertRules.ActionTypes.FirstOrDefault(t => t.Equals(request.Type?.Trim(), StringComparison.OrdinalIgnoreCase));
194 if (type is null)
195 return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.Invalid,
196 "Type must be CreateIntervention, LinkInterventions, NoActionRequired, Reschedule or PutOnHold.");
197
198 lock (_store.Sync)
199 {
200 var alert = _store.Items.FirstOrDefault(a => a.Id == id);
201 if (alert is null) return ServiceOutcome<WeatherAlert>.Fail(OutcomeKind.NotFound, "Weather alert not found.");
202
203 _currentActor = actor;
204 var result = type switch
205 {
206 WeatherAlertRules.CreateIntervention => CreateFrom(alert, request),
207 WeatherAlertRules.LinkInterventions => Link(alert, request),
208 WeatherAlertRules.NoActionRequired => NoAction(request),
209 WeatherAlertRules.Reschedule => Reschedule(alert, request),
210 _ => PutOnHold(alert, request)
211 };
212 if (result.Error is not null) return ServiceOutcome<WeatherAlert>.Fail(result.Kind, result.Error);
213
214 alert.Actions.Add(new WeatherAlertAction
215 {
216 Type = type,
217 InterventionIds = result.Ids,
218 Notes = result.Notes,
219 Actor = actor,
220 TakenAt = DateTime.UtcNow
221 });
222 foreach (var iid in result.Ids.Where(i => !alert.LinkedInterventionIds.Contains(i)))
223 alert.LinkedInterventionIds.Add(iid);
224 _store.Save();
225
226 _audit.Record("WeatherAlert", alert.Id, AuditActions.WeatherAction, actor,
227 $"{type}{(result.Ids.Count > 0 ? " [" + string.Join(", ", result.Ids) + "]" : "")}: {result.Notes}");
228 return ServiceOutcome<WeatherAlert>.Ok(alert);
229 }
230 }

237 private ActionResult CreateFrom(WeatherAlert alert, WeatherAlertActionDto request)
238 {
239 var dto = new CreateInterventionDto
240 {
241 Description = string.IsNullOrWhiteSpace(request.Description) ? $"Weather response: {alert.Title}" : request.Description.Trim(),
242 Location = string.IsNullOrWhiteSpace(request.Location) ? alert.AffectedArea ?? "" : request.Location.Trim(),
243 IssueType = WeatherAlertRules.WeatherIssueType,
244 Priority = request.Priority,
245 Instructions = request.Notes
246 };
247 var errors = InterventionService.Validate(dto);
248 if (errors.Count > 0) return ActionResult.Fail(OutcomeKind.Invalid, string.Join(" ", errors));
249
250 var created = _interventions.Create(dto);
251 return new ActionResult(new List<int> { created.Id }, request.Notes ?? $"Intervention #{created.Id} created.");
252 }

254 private ActionResult Link(WeatherAlert alert, WeatherAlertActionDto request)
255 {
256 var ids = (request.InterventionIds ?? new List<int>()).Distinct().ToList();
257 if (ids.Count == 0) return ActionResult.Fail(OutcomeKind.Invalid, "At least one intervention ID is required.");
258 var missing = ids.Where(i => _interventions.GetById(i) is null).ToList();
259 if (missing.Count > 0) return ActionResult.Fail(OutcomeKind.NotFound, $"Intervention(s) not found: {string.Join(", ", missing)}.");
260 return new ActionResult(ids, request.Notes ?? "Existing interventions linked.");
261 }

263 private static ActionResult NoAction(WeatherAlertActionDto request) =>
264 string.IsNullOrWhiteSpace(request.Notes)
265 ? ActionResult.Fail(OutcomeKind.Invalid, "Notes are required when recording that no action is required.")
266 : new ActionResult(new List<int>(), request.Notes.Trim());

268 private ActionResult Reschedule(WeatherAlert alert, WeatherAlertActionDto request)
269 {
270 if (LinkedTarget(alert, request) is { } error) return error;
271 if (request.PlannedStartDate is null || request.PlannedEndDate is null)
272 return ActionResult.Fail(OutcomeKind.Invalid, "Planned start and end dates are required.");
273 if (request.PlannedEndDate < request.PlannedStartDate)
274 return ActionResult.Fail(OutcomeKind.Invalid, "Planned end date cannot be before the start date.");
275
276 var target = _interventions.GetById(request.InterventionId!.Value)!;
277 var before = target.PlannedStartDate;
278 var previousStatus = target.Status;
279 _interventions.ScheduleIntervention(target.Id, request.PlannedStartDate.Value, request.PlannedEndDate.Value);
280 _audit.Record("Intervention", target.Id, AuditActions.Scheduled, _currentActor,
281 $"{request.PlannedStartDate:yyyy-MM-dd HH:mm} - {request.PlannedEndDate:yyyy-MM-dd HH:mm} (weather alert #{alert.Id})");
282 if (previousStatus != InterventionRules.ScheduledStatus)
283 _audit.Record("Intervention", target.Id, AuditActions.StatusChanged, _currentActor,
284 $"{previousStatus} -> {InterventionRules.ScheduledStatus} (weather alert #{alert.Id})");
285 return new ActionResult(new List<int> { target.Id },
286 request.Notes ?? $"Rescheduled from {before:yyyy-MM-dd HH:mm} to {request.PlannedStartDate:yyyy-MM-dd HH:mm}.");
287 }
288
289 private ActionResult PutOnHold(WeatherAlert alert, WeatherAlertActionDto request)
290 {
291 if (LinkedTarget(alert, request) is { } error) return error;
292 var target = _interventions.GetById(request.InterventionId!.Value)!;
293 if (target.Status is InterventionRules.CompletedStatus or InterventionRules.CancelledStatus)
294 return ActionResult.Fail(OutcomeKind.Conflict, $"Intervention {target.Id} is {target.Status} and cannot be put on hold.");
295
296 var previous = target.Status;
297 _interventions.UpdateStatus(target.Id, "On Hold");
298 _audit.Record("Intervention", target.Id, AuditActions.StatusChanged, _currentActor,
299 $"{previous} -> On Hold (weather alert #{alert.Id})");
300 return new ActionResult(new List<int> { target.Id }, request.Notes ?? $"Status {previous} -> On Hold.");
301 }
302
303 private ActionResult? LinkedTarget(WeatherAlert alert, WeatherAlertActionDto request)
304 {
305 if (request.InterventionId is null) return ActionResult.Fail(OutcomeKind.Invalid, "Intervention ID is required.");
306 if (_interventions.GetById(request.InterventionId.Value) is null)
307 return ActionResult.Fail(OutcomeKind.NotFound, "Intervention not found.");
308 if (!alert.LinkedInterventionIds.Contains(request.InterventionId.Value))
309 return ActionResult.Fail(OutcomeKind.Conflict, "Only interventions linked to this alert can be changed from it. Link it first.");
310 return null;
311 }

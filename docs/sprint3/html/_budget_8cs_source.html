75 /// <summary>
76 /// Returns the reason the status change is blocked, or null when it is allowed (BUD-04).
77 /// </summary>
78 /// <param name="intervention">The intervention.</param>
79 /// <param name="newStatus">Requested status.</param>
80 /// <returns>The value, or null.</returns>
81 public string? BlockReason(Intervention intervention, string newStatus)
82 {
83 if (!RequireApprovalToStart
84 || !string.Equals(newStatus, BudgetRules.InProgressStatus, StringComparison.OrdinalIgnoreCase)
85 || intervention.BudgetStatus == BudgetRules.Approved)
86 {
87 return null;
88 }
89
90 return $"Work cannot start: the budget status is \"{intervention.BudgetStatus}\". " +
91 "The cost estimate must be approved by the Budget service before the intervention can move to In Progress (BUD-04).";
92 }

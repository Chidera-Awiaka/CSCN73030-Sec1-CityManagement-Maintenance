139 {
140 var (service, interventions) = NewServices();

143
144 var outcome = service.Accept(report.Id,

148 Assert.Equal("Accepted", outcome.Report!.Status);
149 Assert.Equal("Agent A", outcome.Report.ReviewedBy);
150 Assert.NotNull(outcome.Report.ReviewedAt);

152
153 var intervention = interventions.GetById(outcome.Report.InterventionId!.Value);
154 Assert.NotNull(intervention);
155 Assert.Equal("Large pothole in the right lane", intervention!.Description);
156 Assert.Equal("King Street North", intervention.Location);
157 Assert.Equal("High", intervention.Priority);
158 Assert.Equal("To Do", intervention.Status);
159 }

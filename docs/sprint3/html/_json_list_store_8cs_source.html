5/// <summary>
6/// Thread-safe list of records saved to a JSON file (or kept in memory when no file
7/// is given, which is how the unit tests use it). Shared by the Sprint 3 services.
8/// </summary>
9/// <typeparam name="T">Record type.</typeparam>
10public class JsonListStore<T>
11{
12 private static readonly JsonSerializerOptions Options = new() { WriteIndented = true };
13
14 private readonly string? _filePath;
15 private readonly List<T> _items;
16
17 /// <summary>Lock that callers hold while reading or changing <see cref="Items"/>.</summary>
18 public object Sync { get; } = new();
19
20 /// <summary>Creates the store and loads any saved records.</summary>
21 /// <param name="filePath">JSON file; null keeps the data in memory only.</param>
22 public JsonListStore(string? filePath)
23 {
24 _filePath = filePath;
25 _items = Load();
26 }
27
28 /// <summary>The records. Only use while holding <see cref="Sync"/>.</summary>
29 public List<T> Items => _items;
30
31 /// <summary>Next numeric ID, given a selector for the existing IDs.</summary>
32 public int NextId(Func<T, int> id) => _items.Count == 0 ? 1 : _items.Max(id) + 1;
33
34 /// <summary>Writes the records to the JSON file (no-op in memory mode).</summary>
35 public void Save()
36 {
37 if (_filePath is null) return;
38 var folder = Path.GetDirectoryName(_filePath);
39 if (!string.IsNullOrEmpty(folder)) Directory.CreateDirectory(folder);
40 File.WriteAllText(_filePath, JsonSerializer.Serialize(_items, Options));
41 }
42
43 private List<T> Load()
44 {
45 if (_filePath is null || !File.Exists(_filePath)) return new List<T>();
46 var json = File.ReadAllText(_filePath);
47 return string.IsNullOrWhiteSpace(json)
48 ? new List<T>()
49 : JsonSerializer.Deserialize<List<T>>(json, Options) ?? new List<T>();
50 }
51}

5/// <summary>

13 /// <summary>

29 /// <inheritdoc />
30 public IReadOnlyList<Property> Search(string? propertyId, string? address)
31 {
32 EnsureAvailable();
33
34 IEnumerable<Property> query = _properties;
35
36 if (!string.IsNullOrWhiteSpace(propertyId))
37 {
38 query = query.Where(p =>
39 string.Equals(p.PropertyId, propertyId.Trim(), StringComparison.OrdinalIgnoreCase));
40 }
41
42 if (!string.IsNullOrWhiteSpace(address))
43 {
44 query = query.Where(p =>
45 p.Address.Contains(address.Trim(), StringComparison.OrdinalIgnoreCase));
46 }
47
48 return query.OrderBy(p => p.PropertyId).ToList();
49 }
50
51 /// <inheritdoc />
52 public Property? GetById(string propertyId)
53 {
54 EnsureAvailable();
55
56 if (string.IsNullOrWhiteSpace(propertyId))
57 {
58 return null;
59 }
60
61 return _properties.FirstOrDefault(p =>
62 string.Equals(p.PropertyId, propertyId.Trim(), StringComparison.OrdinalIgnoreCase));
63 }

65 /// <inheritdoc />

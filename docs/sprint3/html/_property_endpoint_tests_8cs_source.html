30 /// <summary>Test: search by Property ID returns 200 with ID and address.</summary>

75 /// <summary>Test: a linked property is visible in the intervention details.</summary>
76 [Fact]
77 public void Link_ThenVisibleInDetails()
78 {
79 var controller = Interventions();
80
81 Assert.Equal(200, StatusOf(controller.LinkProperty(1, new LinkPropertyDto { PropertyId = "PR-1003" })));
82
83 var details = Assert.IsType<OkObjectResult>(controller.GetLinkedProperties(1));
84 var properties = Assert.IsType<List<Property>>(details.Value);
85 Assert.Contains(properties, p => p.PropertyId == "PR-1003" && p.Address.Contains("Victoria"));
86 }

24 /// <summary>
25 /// Sends a road closure, update or reopening (ESV-06 / SEC-03 / TRP-02) to E-services, Security and
26 /// Public Transport (or the services listed). 200 with the notice and per-service delivery results.
27 /// </summary>
28 /// <param name="id">Intervention ID.</param>
29 /// <param name="request">Closure status, location, dates and services to notify.</param>
30 /// <returns>The HTTP result (status codes described above).</returns>
31 [HttpPost("api/interventions/{id:int}/road-closures")]
32 public IActionResult RoadClosure(int id, RoadClosureDto request) =>
33 Result(_service.SendRoadClosure(id, request, RequestContext.Actor(Request)));
34
35 /// <summary>Sends the current work status to Security (or the services listed) (SEC-03).</summary>
36 /// <param name="id">Intervention ID.</param>
37 /// <param name="request">Services to notify (optional).</param>
38 /// <returns>The HTTP result (status codes described above).</returns>
39 [HttpPost("api/interventions/{id:int}/work-status-updates")]
40 public IActionResult WorkStatus(int id, WorkStatusUpdateDto? request) =>
41 Result(_service.SendWorkStatus(id, request, RequestContext.Actor(Request)));
42
43 /// <summary>Sends a property-related service disruption to the Property Registry (REG-04).</summary>
44 /// <param name="id">Intervention ID.</param>
45 /// <param name="request">Disruption type, update type and period.</param>
46 /// <returns>The HTTP result (status codes described above).</returns>
47 [HttpPost("api/interventions/{id:int}/property-disruptions")]
48 public IActionResult PropertyDisruption(int id, PropertyDisruptionDto request) =>
49 Result(_service.SendPropertyDisruption(id, request, RequestContext.Actor(Request)));
50
51 /// <summary>Notices sent for an intervention, with delivery status.</summary>
52 /// <param name="id">Intervention ID.</param>
53 /// <param name="kind">Optional notice kind: RoadClosure, WorkStatusUpdate or PropertyDisruption.</param>
54 /// <returns>The HTTP result (status codes described above).</returns>
55 [HttpGet("api/interventions/{id:int}/notices")]
56 public IActionResult ForIntervention(int id, [FromQuery] string? kind) => Ok(_service.GetAll(id, kind));
57
58 /// <summary>All notices, optionally by kind (RoadClosure, WorkStatusUpdate, PropertyDisruption).</summary>
59 /// <param name="kind">Optional notice kind: RoadClosure, WorkStatusUpdate or PropertyDisruption.</param>
60 /// <returns>The HTTP result (status codes described above).</returns>
61 [HttpGet("api/notices")]
62 public IActionResult All([FromQuery] string? kind) => Ok(_service.GetAll(null, kind));
63
64 private IActionResult Result(ServiceOutcome<Models.OutboundNotice> outcome) =>
65 this.ToResult(outcome, n => Ok(new
66 {
67 duplicate = outcome.Duplicate,
68 allDelivered = n.Deliveries.All(d => d.Status == Models.ExchangeRules.Succeeded),
69 notice = n
70 }));
71}

6/// <summary>
7/// Records and queries the audit history (TEC-05). Status changes, weather-response
8/// decisions, assignments and received budget decisions keep their date and their
9/// actor (a Maintenance user) or source (an external service).
10/// </summary>
11public class AuditService
12{
13 private readonly JsonListStore<AuditEntry> _store;
14
15 /// <summary>Creates the service.</summary>
16 /// <param name="filePath">JSON file, or null for memory only (tests).</param>
17 public AuditService(string? filePath = null)
18 {
19 _store = new JsonListStore<AuditEntry>(filePath);
20 }
21
22 /// <summary>Adds an entry made by a Maintenance user.</summary>
23 /// <param name="entityType">Kind of record, e.g. Intervention.</param>
24 /// <param name="entityId">Record ID.</param>
25 /// <param name="action">What happened (see <see cref="AuditActions"/>).</param>
26 /// <param name="actor">User who made the change.</param>
27 /// <param name="details">Readable details.</param>
28 /// <returns>The stored entry.</returns>

31
32 /// <summary>Adds an entry caused by an external service (no Maintenance user involved).</summary>
33 /// <param name="entityType">Kind of record.</param>
34 /// <param name="entityId">Record ID.</param>
35 /// <param name="action">What happened.</param>
36 /// <param name="source">External service, e.g. Budget.</param>
37 /// <param name="details">Readable details.</param>
38 /// <returns>The stored entry.</returns>
39 public AuditEntry RecordFromService(string entityType, object entityId, string action, string source, string details) =>
40 Add(entityType, entityId, action, null, source, details);
41
42 /// <summary>Lists entries, newest first, optionally for one record.</summary>
43 /// <param name="entityType">Optional record kind.</param>
44 /// <param name="entityId">Optional record ID.</param>
45 /// <returns>Matching entries.</returns>
46 public List<AuditEntry> Query(string? entityType = null, string? entityId = null)
47 {
48 lock (_store.Sync)
49 {
50 return _store.Items
51 .Where(x => string.IsNullOrWhiteSpace(entityType)
52 || string.Equals(x.EntityType, entityType, StringComparison.OrdinalIgnoreCase))
53 .Where(x => string.IsNullOrWhiteSpace(entityId) || x.EntityId == entityId)
54 .OrderByDescending(x => x.Timestamp).ThenByDescending(x => x.Id)
55 .ToList();
56 }
57 }
58
59 private AuditEntry Add(string entityType, object entityId, string action, string? actor, string source, string details)
60 {
61 lock (_store.Sync)
62 {
63 var entry = new AuditEntry
64 {
65 Id = _store.NextId(x => x.Id),
66 Timestamp = DateTime.UtcNow,
67 EntityType = entityType,
68 EntityId = entityId.ToString() ?? "",
69 Action = action,
70 Actor = actor,
71 Source = source,
72 Details = details
73 };
74 _store.Items.Add(entry);
75 _store.Save();
76 return entry;
77 }
78 }
79}

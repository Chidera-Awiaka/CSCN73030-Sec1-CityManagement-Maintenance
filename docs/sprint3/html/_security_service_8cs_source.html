9/// <summary>
10/// Security integration.
11/// SEC-01 send an assistance request (reason, location, period, intervention ID; transmission status visible).
12/// SEC-02 receive Security's response (linked to the request; outcome and comments visible). Only the
13/// Security service can send a response, so Maintenance users cannot approve on Security's behalf.
14/// </summary>
15public class SecurityService
16{
17 private const string Operation = "AssistanceRequest";
18 private readonly JsonListStore<SecurityRequest> _store;
19 private readonly InterventionService _interventions;
20 private readonly ExchangeService _exchanges;
21 private readonly AuditService _audit;
22
23 /// <summary>Creates the service.</summary>
24 /// <param name="interventions">Intervention service.</param>
25 /// <param name="exchanges">Exchange log (TEC-04).</param>
26 /// <param name="audit">Audit history (TEC-05).</param>
27 /// <param name="filePath">JSON file used for storage, or null to keep data in memory (tests).</param>
28 public SecurityService(InterventionService interventions, ExchangeService exchanges, AuditService audit, string? filePath = null)
29 {
30 _interventions = interventions;
31 _exchanges = exchanges;
32 _audit = audit;
33 _store = new JsonListStore<SecurityRequest>(filePath);
34 _exchanges.Succeeded += OnExchangeSucceeded;
35 }
36
37 /// <summary>Lists requests, newest first, optionally for one intervention.</summary>
38 /// <param name="interventionId">Intervention ID.</param>
39 /// <returns>The matching records.</returns>
40 public List<SecurityRequest> GetAll(int? interventionId = null)
41 {
42 lock (_store.Sync)
43 {
44 return _store.Items.Where(r => interventionId is null || r.InterventionId == interventionId)
45 .OrderByDescending(r => r.CreatedAt).ThenByDescending(r => r.Id).ToList();
46 }
47 }

49 /// <summary>Creates and sends an assistance request (SEC-01). A failed send is kept and can be retried.</summary>
50 /// <param name="interventionId">Intervention ID.</param>
51 /// <param name="request">Reason, location and period.</param>
52 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
53 /// <returns>Success with the record, or the failure kind and messages.</returns>
54 public ServiceOutcome<SecurityRequest> Request(int interventionId, SecurityRequestDto request, string actor)
55 {
56 if (request is null) return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Invalid, "Request body is required.");
57 var intervention = _interventions.GetById(interventionId);
58 if (intervention is null) return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.NotFound, "Intervention not found.");
59
60 var errors = new List<string>();
61 if (string.IsNullOrWhiteSpace(request.Reason)) errors.Add("Reason is required.");
62 if (request.PeriodStart is null) errors.Add("Period start is required.");
63 if (request.PeriodEnd is null) errors.Add("Period end is required.");
64 if (request.PeriodStart is not null && request.PeriodEnd is not null && request.PeriodEnd < request.PeriodStart)
65 errors.Add("Period end cannot be before period start.");
66 var location = string.IsNullOrWhiteSpace(request.Location) ? intervention.Location : request.Location.Trim();
67 if (string.IsNullOrWhiteSpace(location)) errors.Add("Location is required.");
68 if (errors.Count > 0) return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Invalid, errors);
69
70 SecurityRequest record;
71 lock (_store.Sync)
72 {
73 record = new SecurityRequest
74 {
75 Id = _store.NextId(r => r.Id),
76 InterventionId = interventionId,
77 Reason = request.Reason.Trim(),
78 Location = location,
79 PeriodStart = request.PeriodStart!.Value,
80 PeriodEnd = request.PeriodEnd!.Value,
81 RequestedBy = actor,
82 CreatedAt = DateTime.UtcNow
83 };
84 _store.Items.Add(record);
85 _store.Save();
86 }
87
88 var payload = JsonSerializer.Serialize(new
89 {
90 maintenanceRequestId = record.Id,
91 interventionId,
92 reason = record.Reason,
93 location = record.Location,
94 periodStart = record.PeriodStart,
95 periodEnd = record.PeriodEnd,
96 requestedBy = actor
97 });
98 var sent = _exchanges.Send(ExternalServiceNames.Security, Operation, $"security-request:{record.Id}", payload, interventionId);
99
100 lock (_store.Sync)
101 {
102 record.ExchangeId = sent.Exchange.Id;
103 record.TransmissionStatus = sent.Exchange.Status;
104 record.TransmissionError = sent.Exchange.LastError;
105 record.SecurityReference = sent.Exchange.ExternalReference;
106 _store.Save();
107 }
108 return ServiceOutcome<SecurityRequest>.Ok(record);
109 }
110
111 /// <summary>
112 /// Receives Security's response (SEC-02). Rejected (403) unless the caller is the Security
113 /// service; a repeated delivery is ignored (TEC-04).
114 /// </summary>
115 /// <param name="response">Response message from Security.</param>
116 /// <param name="sourceService">Value of the X-Source-Service header (the calling service).</param>
117 /// <returns>Success with the record, or the failure kind and messages.</returns>
118 public ServiceOutcome<SecurityRequest> ReceiveResponse(SecurityResponseDto response, string? sourceService)
119 {
120 if (!string.Equals(sourceService, ExternalServiceNames.Security, StringComparison.OrdinalIgnoreCase))
121 return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Forbidden,
122 "Only the Security service can respond to assistance requests (X-Source-Service: Security). Maintenance users cannot approve requests on Security's behalf.");
123
124 if (response is null) return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Invalid, "Request body is required.");
125 var errors = new List<string>();
126 if (string.IsNullOrWhiteSpace(response.ResponseId)) errors.Add("Response ID is required.");
127 if (string.IsNullOrWhiteSpace(response.RequestReference)) errors.Add("Request reference is required.");
128 var outcome = SecurityRules.Outcomes.FirstOrDefault(o => o.Equals(response.Outcome?.Trim(), StringComparison.OrdinalIgnoreCase));
129 if (outcome is null) errors.Add("Outcome must be Approved, Declined or Partially Approved.");
130 if (errors.Count > 0) return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Invalid, errors);
131
132 SecurityRequest? record;
133 lock (_store.Sync)
134 {
135 record = _store.Items.FirstOrDefault(r => r.SecurityReference == response.RequestReference.Trim());
136 }
137 if (record is null)
138 return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.NotFound, $"No assistance request with reference {response.RequestReference}.");
139
140 // Recorded outside this service's lock (the exchange log raises events that take it).
141 var (_, duplicate) = _exchanges.RecordInbound(ExternalServiceNames.Security, "AssistanceResponse",
142 response.ResponseId.Trim(), JsonSerializer.Serialize(response), record.InterventionId);
143 if (duplicate) return ServiceOutcome<SecurityRequest>.Ok(record, duplicate: true);
144
145 lock (_store.Sync)
146 {
147 if (record.Response is not null)
148 return ServiceOutcome<SecurityRequest>.Fail(OutcomeKind.Conflict, "Security has already responded to this request.");
149
150 record.Response = new SecurityResponse
151 {
152 Outcome = outcome!,
153 Comments = response.Comments?.Trim(),
154 RespondedBy = response.RespondedBy?.Trim(),
155 ResponseId = response.ResponseId.Trim(),
156 ReceivedAt = DateTime.UtcNow
157 };
158 _store.Save();
159 }
160 _audit.RecordFromService("SecurityRequest", record.Id, AuditActions.SecurityResponseReceived, ExternalServiceNames.Security,
161 $"{outcome} for intervention {record.InterventionId}: {response.Comments}");
162 return ServiceOutcome<SecurityRequest>.Ok(record);
163 }
164
165 private void OnExchangeSucceeded(ExternalExchange exchange)
166 {
167 if (exchange.Operation != Operation) return;
168 lock (_store.Sync)
169 {
170 var record = _store.Items.FirstOrDefault(r => r.ExchangeId == exchange.Id);
171 if (record is null) return;
172 record.TransmissionStatus = exchange.Status;
173 record.TransmissionError = null;
174 record.SecurityReference = exchange.ExternalReference;
175 _store.Save();
176 }
177 }
178}

9/// <summary>
10/// Sends notices about interventions to other services and records every delivery (TEC-04).
11/// ESV-06 road closure / reopening to E-services; TRP-02 road closures and changes to Public Transport;
12/// SEC-03 road closures and work-status updates to Security; REG-04 property-related service
13/// disruptions to the Property Registry. Sending the same notice twice does not create a duplicate.
14/// </summary>

32 /// <summary>Lists notices, newest first, with delivery status refreshed from the exchange log.</summary>
33 /// <param name="interventionId">Intervention ID.</param>
34 /// <param name="kind">Optional notice kind: RoadClosure, WorkStatusUpdate or PropertyDisruption.</param>
35 /// <returns>The matching records.</returns>
36 public List<OutboundNotice> GetAll(int? interventionId = null, string? kind = null)
37 {
38 List<OutboundNotice> notices;
39 lock (_store.Sync)
40 {
41 notices = _store.Items
42 .Where(n => interventionId is null || n.InterventionId == interventionId)
43 .Where(n => string.IsNullOrWhiteSpace(kind) || string.Equals(n.Kind, kind, StringComparison.OrdinalIgnoreCase))
44 .OrderByDescending(n => n.CreatedAt).ThenByDescending(n => n.Id)
45 .ToList();
46 }
47 foreach (var notice in notices) Refresh(notice);
48 return notices;
49 }

51 /// <summary>Sends a road closure / update / reopening (ESV-06, SEC-03, TRP-02).</summary>
52 /// <param name="interventionId">Intervention ID.</param>
53 /// <param name="request">Closure status, location, dates and services to notify.</param>
54 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
55 /// <returns>Success with the record, or the failure kind and messages.</returns>
56 public ServiceOutcome<OutboundNotice> SendRoadClosure(int interventionId, RoadClosureDto request, string actor)
57 {
58 if (request is null) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.Invalid, "Request body is required.");
59 var intervention = _interventions.GetById(interventionId);
60 if (intervention is null) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.NotFound, "Intervention not found.");
61
62 var errors = new List<string>();
63 var status = NoticeRules.ClosureStatuses.FirstOrDefault(s => s.Equals(request.ClosureStatus?.Trim(), StringComparison.OrdinalIgnoreCase));
64 if (status is null) errors.Add("Closure status must be Closed, Updated or Reopened.");
65 var location = string.IsNullOrWhiteSpace(request.Location) ? intervention.Location : request.Location.Trim();
66 if (string.IsNullOrWhiteSpace(location)) errors.Add("Location is required.");
67 var start = request.PlannedStart ?? intervention.PlannedStartDate;
68 var end = request.PlannedEnd ?? intervention.PlannedEndDate;
69 if (start is null) errors.Add("Planned start date is required (schedule the intervention or supply it).");
70 if (start is not null && end is not null && end < start) errors.Add("Planned end cannot be before planned start.");
71 var services = Services(request.Services, NoticeRules.RoadClosureServices, errors);
72 if (errors.Count > 0) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.Invalid, errors);
73
74 var payload = new
75 {
76 interventionId,
77 closureStatus = status,
78 location,
79 plannedStart = start,
80 plannedEnd = end,
81 interventionStatus = intervention.Status
82 };
83 var key = $"{interventionId}:{status}:{location}:{start:O}:{end:O}";
84 var summary = $"{status} - {location}, {start:yyyy-MM-dd HH:mm}{(end is null ? "" : $" to {end:yyyy-MM-dd HH:mm}")}";
85 return Send(NoticeRules.RoadClosure, "RoadClosureNotice", interventionId, key, summary, payload, services, actor);
86 }

115 /// <summary>
116 /// Sends a property-related service disruption to the Property Registry (REG-04): property IDs,
117 /// disruption type, applicable period and linked intervention; updates and restorations can be sent.
118 /// </summary>
119 /// <param name="interventionId">Intervention ID.</param>
120 /// <param name="request">Disruption type, update type and period.</param>
121 /// <param name="actor">Maintenance user making the change (recorded in the audit history).</param>
122 /// <returns>Success with the record, or the failure kind and messages.</returns>
123 public ServiceOutcome<OutboundNotice> SendPropertyDisruption(int interventionId, PropertyDisruptionDto request, string actor)
124 {
125 if (request is null) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.Invalid, "Request body is required.");
126 var intervention = _interventions.GetById(interventionId);
127 if (intervention is null) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.NotFound, "Intervention not found.");
128
129 var errors = new List<string>();
130 if (intervention.LinkedPropertyIds.Count == 0) errors.Add("Link at least one property to the intervention first (REG-02).");
131 var type = NoticeRules.DisruptionTypes.FirstOrDefault(t => t.Equals(request.DisruptionType?.Trim(), StringComparison.OrdinalIgnoreCase));
132 if (type is null) errors.Add("Disruption type must be Water, Electricity, Gas, Access or Other.");
133 var update = string.IsNullOrWhiteSpace(request.UpdateType) ? "New"
134 : NoticeRules.DisruptionUpdateTypes.FirstOrDefault(t => t.Equals(request.UpdateType.Trim(), StringComparison.OrdinalIgnoreCase));
135 if (update is null) errors.Add("Update type must be New, Update or Restored.");
136 if (request.PeriodStart is null) errors.Add("Period start is required.");
137 if (request.PeriodStart is not null && request.PeriodEnd is not null && request.PeriodEnd < request.PeriodStart)
138 errors.Add("Period end cannot be before period start.");
139 if (errors.Count > 0) return ServiceOutcome<OutboundNotice>.Fail(OutcomeKind.Invalid, errors);
140
141 var payload = new
142 {
143 interventionId,
144 propertyIds = intervention.LinkedPropertyIds,
145 disruptionType = type,
146 updateType = update,
147 periodStart = request.PeriodStart,
148 periodEnd = request.PeriodEnd,
149 notes = request.Notes?.Trim()
150 };
151 var key = $"{interventionId}:{type}:{update}:{string.Join(",", intervention.LinkedPropertyIds)}:{request.PeriodStart:O}:{request.PeriodEnd:O}";
152 var summary = $"{update} {type} disruption - {string.Join(", ", intervention.LinkedPropertyIds)}";
153 return Send(NoticeRules.PropertyDisruption, "PropertyDisruption", interventionId, key, summary, payload,
154 new List<string> { ExternalServiceNames.PropertyRegistration }, actor);
155 }

157 private static List<string> Services(List<string>? requested, IEnumerable<string> defaults, List<string> errors)
158 {
159 if (requested is null || requested.Count == 0) return defaults.ToList();
160 var result = new List<string>();
161 foreach (var name in requested.Where(n => !string.IsNullOrWhiteSpace(n)).Distinct(StringComparer.OrdinalIgnoreCase))
162 {
163 var match = ExternalServiceNames.Outbound.FirstOrDefault(s => s.Equals(name.Trim(), StringComparison.OrdinalIgnoreCase));
164 if (match is null) errors.Add($"Unknown service: {name}.");
165 else result.Add(match);
166 }
167 if (result.Count == 0 && errors.Count == 0) errors.Add("At least one service is required.");
168 return result;
169 }

171 private ServiceOutcome<OutboundNotice> Send(string kind, string operation, int interventionId, string key, string summary,
172 object payload, List<string> services, string actor)
173 {
174 var payloadJson = JsonSerializer.Serialize(payload);
175 lock (_store.Sync)
176 {
177 var existing = _store.Items.FirstOrDefault(n => n.Kind == kind && n.InterventionId == interventionId && n.PayloadJson == payloadJson
178 && services.All(s => n.Deliveries.Any(d => d.Service == s)));
179 if (existing is not null && existing.Deliveries.All(d => _exchanges.GetById(d.ExchangeId)?.Status == ExchangeRules.Succeeded))
180 {
181 Refresh(existing);
182 return ServiceOutcome<OutboundNotice>.Ok(existing, duplicate: true);
183 }
184 }
185
186 var deliveries = services.Select(service =>
187 {
188 var sent = _exchanges.Send(service, operation, $"{kind}:{key}", payloadJson, interventionId);
189 return new NoticeDelivery
190 {
191 Service = service,
192 ExchangeId = sent.Exchange.Id,
193 Status = sent.Exchange.Status,
194 Reference = sent.Exchange.ExternalReference,
195 Error = sent.Exchange.LastError
196 };
197 }).ToList();
198
199 lock (_store.Sync)
200 {
201 var notice = _store.Items.FirstOrDefault(n => n.Kind == kind && n.InterventionId == interventionId && n.PayloadJson == payloadJson);
202 if (notice is null)
203 {
204 notice = new OutboundNotice
205 {
206 Id = _store.NextId(n => n.Id),
207 Kind = kind,
208 InterventionId = interventionId,
209 Summary = summary,
210 PayloadJson = payloadJson,
211 SentBy = actor,
212 CreatedAt = DateTime.UtcNow
213 };
214 _store.Items.Add(notice);
215 }
216 foreach (var d in deliveries)
217 {
218 notice.Deliveries.RemoveAll(x => x.Service == d.Service);
219 notice.Deliveries.Add(d);
220 }
221 _store.Save();
222 return ServiceOutcome<OutboundNotice>.Ok(notice);
223 }
224 }

226 private void Refresh(OutboundNotice notice)
227 {
228 foreach (var d in notice.Deliveries)
229 {
230 var exchange = _exchanges.GetById(d.ExchangeId);
231 if (exchange is null) continue;
232 d.Status = exchange.Status;
233 d.Reference = exchange.ExternalReference;
234 d.Error = exchange.LastError;
235 }
236 }

9/// <summary>
10/// Route changes and diversions received from Public Transport (TRP-03): external references
11/// retained, affected routes and effective dates shown, and they can be linked to interventions.
12/// Only the Public Transport service can send them; a repeated delivery is ignored (TEC-04).
13/// </summary>
14public class TransitDiversionService
15{
16 private readonly JsonListStore<RouteDiversion> _store;
17 private readonly InterventionService _interventions;
18 private readonly ExchangeService _exchanges;
19
20 /// <summary>Creates the service.</summary>
21 /// <param name="interventions">Intervention service.</param>
22 /// <param name="exchanges">Exchange log (TEC-04).</param>
23 /// <param name="filePath">JSON file used for storage, or null to keep data in memory (tests).</param>
24 public TransitDiversionService(InterventionService interventions, ExchangeService exchanges, string? filePath = null)
25 {
26 _interventions = interventions;
27 _exchanges = exchanges;
28 _store = new JsonListStore<RouteDiversion>(filePath);
29 }
30
31 /// <summary>Lists diversions, newest effective date first, optionally those linked to an intervention.</summary>
32 /// <param name="interventionId">Intervention ID.</param>
33 /// <returns>The matching records.</returns>
34 public List<RouteDiversion> GetAll(int? interventionId = null)
35 {
36 lock (_store.Sync)
37 {
38 return _store.Items.Where(d => interventionId is null || d.LinkedInterventionIds.Contains(interventionId.Value))
39 .OrderByDescending(d => d.EffectiveFrom).ThenByDescending(d => d.Id).ToList();
40 }
41 }

43 /// <summary>Receives a route change / diversion (TRP-03).</summary>
44 /// <param name="request">Route-change message sent by Public Transport.</param>
45 /// <param name="sourceService">Value of the X-Source-Service header (the calling service).</param>
46 /// <returns>Success with the record, or the failure kind and messages.</returns>
47 public ServiceOutcome<RouteDiversion> Receive(RouteDiversionDto request, string? sourceService)
48 {
49 if (!string.Equals(sourceService, ExternalServiceNames.PublicTransport, StringComparison.OrdinalIgnoreCase))
50 return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.Forbidden, "Only the Public Transport service can send route changes (X-Source-Service: Public Transport).");
51 if (request is null) return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.Invalid, "Request body is required.");
52
53 var errors = new List<string>();
54 if (string.IsNullOrWhiteSpace(request.ExternalReference)) errors.Add("External reference is required.");
55 var routes = (request.AffectedRoutes ?? new List<string>()).Where(r => !string.IsNullOrWhiteSpace(r)).Select(r => r.Trim()).Distinct().ToList();
56 if (routes.Count == 0) errors.Add("At least one affected route is required.");
57 if (string.IsNullOrWhiteSpace(request.Description)) errors.Add("Description is required.");
58 if (request.EffectiveFrom is null) errors.Add("Effective-from date is required.");
59 if (request.EffectiveFrom is not null && request.EffectiveTo is not null && request.EffectiveTo < request.EffectiveFrom)
60 errors.Add("Effective-to cannot be before effective-from.");
61 if (errors.Count > 0) return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.Invalid, errors);
62
63 var reference = request.ExternalReference.Trim();
64 var (_, duplicate) = _exchanges.RecordInbound(ExternalServiceNames.PublicTransport, "RouteDiversion", reference, JsonSerializer.Serialize(request));
65 lock (_store.Sync)
66 {
67 var existing = _store.Items.FirstOrDefault(d => d.ExternalReference == reference);
68 if (existing is not null) return ServiceOutcome<RouteDiversion>.Ok(existing, duplicate: true);
69
70 var diversion = new RouteDiversion
71 {
72 Id = _store.NextId(d => d.Id),
73 ExternalReference = reference,
74 AffectedRoutes = routes,
75 Description = request.Description.Trim(),
76 EffectiveFrom = request.EffectiveFrom!.Value,
77 EffectiveTo = request.EffectiveTo,
78 ReceivedAt = DateTime.UtcNow
79 };
80 _store.Items.Add(diversion);
81 _store.Save();
82 return ServiceOutcome<RouteDiversion>.Ok(diversion);
83 }
84 }
85
86 /// <summary>Links a diversion to an intervention (TRP-03).</summary>
87 /// <param name="id">Route change ID.</param>
88 /// <param name="interventionId">Intervention ID.</param>
89 /// <returns>Success with the record, or the failure kind and messages.</returns>
90 public ServiceOutcome<RouteDiversion> Link(int id, int interventionId)
91 {
92 if (_interventions.GetById(interventionId) is null)
93 return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.NotFound, "Intervention not found.");
94 lock (_store.Sync)
95 {
96 var d = _store.Items.FirstOrDefault(x => x.Id == id);
97 if (d is null) return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.NotFound, "Route change not found.");
98 if (d.LinkedInterventionIds.Contains(interventionId))
99 return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.Conflict, "Already linked to this intervention.");
100 d.LinkedInterventionIds.Add(interventionId);
101 _store.Save();
102 return ServiceOutcome<RouteDiversion>.Ok(d);
103 }
104 }
105
106 /// <summary>Removes a link (TRP-03).</summary>
107 /// <param name="id">Route change ID.</param>
108 /// <param name="interventionId">Intervention ID.</param>
109 /// <returns>Success with the record, or the failure kind and messages.</returns>
110 public ServiceOutcome<RouteDiversion> Unlink(int id, int interventionId)
111 {
112 lock (_store.Sync)
113 {
114 var d = _store.Items.FirstOrDefault(x => x.Id == id);
115 if (d is null) return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.NotFound, "Route change not found.");
116 if (!d.LinkedInterventionIds.Remove(interventionId))
117 return ServiceOutcome<RouteDiversion>.Fail(OutcomeKind.NotFound, "Not linked to this intervention.");
118 _store.Save();
119 return ServiceOutcome<RouteDiversion>.Ok(d);
120 }
121 }
122}

29/// <summary>
30/// Client for sending a message to one external service. The mock below is used until
31/// each team publishes its REST API; a real HTTP client implements the same interface
32/// and follows the contracts in docs/integration-contracts.md (TEC-02, TEC-03).
33/// </summary>
34public interface IExternalServiceClient
35{

39 /// <summary>Sends one message.</summary>

41 /// <param name="payloadJson">Message body (JSON).</param>
42 /// <returns>Success with the service's reference, or failure with an error.</returns>

53/// <summary>
54/// Mock external service (TEC-03). Accepts messages and returns a reference such as
55/// "BUD-1001". Set <see cref="IsAvailable"/> to false to test failure and retry paths.
56/// </summary>
57public class MockExternalService : IExternalServiceClient
58{
59 private readonly object _lock = new();
60 private readonly List<MockMessage> _received = new();
61 private readonly string _prefix;
62 private int _next = 1000;
63
64 /// <summary>Creates a mock.</summary>
65 /// <param name="serviceName">Service name.</param>
66 /// <param name="referencePrefix">Prefix of the references it returns.</param>
67 public MockExternalService(string serviceName, string referencePrefix)
68 {
69 ServiceName = serviceName;
70 _prefix = referencePrefix;
71 }
72
73 /// <inheritdoc />
74 public string ServiceName { get; }
75
76 /// <summary>When false, every call fails as if the service were down.</summary>
77 public bool IsAvailable { get; set; } = true;
78
79 /// <summary>Messages received so far.</summary>
80 public IReadOnlyList<MockMessage> Received
81 {
82 get { lock (_lock) { return _received.ToList(); } }
83 }
84
85 /// <inheritdoc />
86 public ExternalCallResult Send(string operation, string payloadJson)
87 {
88 lock (_lock)
89 {
90 if (!IsAvailable)
91 {
92 return new ExternalCallResult(false, null, $"{ServiceName} is unavailable (mock).");
93 }
94
95 var reference = $"{_prefix}-{++_next}";
96 _received.Add(new MockMessage(operation, payloadJson, reference, DateTime.UtcNow));
97 return new ExternalCallResult(true, reference, null);
98 }
99 }
100}

102/// <summary>Looks up the client for each external service (TEC-03).</summary>
103public class ExternalServiceHub
104{
105 private readonly Dictionary<string, IExternalServiceClient> _clients;
106
107 /// <summary>Creates the hub.</summary>
108 /// <param name="clients">One client per service.</param>
109 public ExternalServiceHub(IEnumerable<IExternalServiceClient> clients)
110 {
111 _clients = clients.ToDictionary(c => c.ServiceName, StringComparer.OrdinalIgnoreCase);
112 }
113
114 /// <summary>Creates a hub with a mock for every outbound service.</summary>
115 /// <returns>The result.</returns>
116 public static ExternalServiceHub WithMocks() => new(new IExternalServiceClient[]
117 {
118 new MockExternalService(ExternalServiceNames.EServices, "ESV"),
119 new MockExternalService(ExternalServiceNames.PropertyRegistration, "REG"),
120 new MockExternalService(ExternalServiceNames.Budget, "BUD"),
121 new MockExternalService(ExternalServiceNames.Security, "SEC"),
122 new MockExternalService(ExternalServiceNames.PublicTransport, "TRP")
123 });
124
125 /// <summary>The client for a service, or null when unknown.</summary>
126 /// <param name="serviceName">The serviceName.</param>
127 /// <returns>The result.</returns>
128 public IExternalServiceClient? Get(string serviceName) =>
129 _clients.TryGetValue(serviceName, out var client) ? client : null;
130
131 /// <summary>All clients.</summary>
132 public IReadOnlyCollection<IExternalServiceClient> All => _clients.Values;
133}

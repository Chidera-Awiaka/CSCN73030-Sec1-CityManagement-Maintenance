23 /// <summary>
24 /// Searches properties by Property ID and/or address (REG-01).
25 /// Example: GET /api/properties?propertyId=PR-1001 or GET /api/properties?address=King
26 /// </summary>
27 /// <param name="propertyId">Exact Property ID (case-insensitive).</param>
28 /// <param name="address">Part of the address (case-insensitive).</param>
29 /// <returns>
30 /// 200 with the matching properties; 400 when no search term is given;
31 /// 404 when nothing matches; 503 when the Property Registry is unavailable.
32 /// </returns>
33 [HttpGet]
34 public IActionResult Search([FromQuery] string? propertyId, [FromQuery] string? address)
35 {
36 if (string.IsNullOrWhiteSpace(propertyId) && string.IsNullOrWhiteSpace(address))
37 {
38 return BadRequest(new { message = "Enter a Property ID or an address to search." });
39 }
40
41 try
42 {
43 var results = _registry.Search(propertyId, address);
44 if (results.Count == 0)
45 {
46 return NotFound(new { message = "No property found matching the search." });
47 }
48
49 return Ok(results);
50 }
51 catch (PropertyRegistryUnavailableException ex)
52 {
53 return StatusCode(StatusCodes.Status503ServiceUnavailable, new { message = ex.Message });
54 }
55 }
56
57 /// <summary>
58 /// Utility information supplied by the Property Registry for one property (REG-03).
59 /// Water, electricity and gas are listed; anything not supplied is named in "missingUtilities".
60 /// </summary>
61 /// <param name="propertyId">Property ID, e.g. PR-1001.</param>
62 /// <returns>200 with the utilities; 404 unknown property; 503 registry unavailable.</returns>
63 [HttpGet("{propertyId}/utilities")]
64 public IActionResult Utilities(string propertyId)
65 {
66 try
67 {
68 var utilities = _registry.GetUtilities(propertyId);
69 if (utilities is null)
70 {
71 return NotFound(new { message = $"Property '{propertyId}' was not found in the Property Registry." });
72 }
73
74 return Ok(new
75 {
76 utilities.PropertyId,
77 water = utilities.Water,
78 electricity = utilities.Electricity,
79 gas = utilities.Gas,
80 missingUtilities = utilities.MissingUtilities,
81 message = utilities.MissingUtilities.Count == 0
82 ? null
83 : $"Not supplied by the Property Registry: {string.Join(", ", utilities.MissingUtilities)}."
84 });
85 }
86 catch (PropertyRegistryUnavailableException ex)
87 {
88 return StatusCode(StatusCodes.Status503ServiceUnavailable, new { message = ex.Message });
89 }
90 }
91}

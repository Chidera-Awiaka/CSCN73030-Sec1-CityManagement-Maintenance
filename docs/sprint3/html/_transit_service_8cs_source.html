56 /// <inheritdoc />
57 public IReadOnlyList<TransitRoute> GetRoutes(string? street = null)
58 {
59 if (!IsAvailable) throw new TransitUnavailableException("The Public Transport service is currently unavailable.");
60
61 return _routes
62 .Where(r => string.IsNullOrWhiteSpace(street)
63 || r.Streets.Any(s => s.Contains(street.Trim(), StringComparison.OrdinalIgnoreCase)
